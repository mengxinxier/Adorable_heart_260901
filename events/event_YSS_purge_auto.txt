# 大清洗善后国策自动推进事件链（替代原 on_daily_YSS 每日倒计时）
add_namespace = YSS_purge_aftermath

# 第 1 步（入口后第 70 天）：完成「YSS_dismiss_purge_directors」并推进下一环
country_event = {
	id = YSS_purge_aftermath.1
	hidden = yes
	is_triggered_only = yes
	fire_only_once = yes

	immediate = {
		if = {
			limit = {
				has_country_flag = YSS_daqingxi_jiaoting
				NOT = { has_completed_focus = YSS_redefine_party_spirit }
			}
			if = {
				limit = { NOT = { has_completed_focus = YSS_emergency_halt_to_purge } }
				complete_national_focus = YSS_emergency_halt_to_purge
			}
			if = {
				limit = {
					has_completed_focus = YSS_emergency_halt_to_purge
					NOT = { has_completed_focus = YSS_dismiss_purge_directors }
				}
				complete_national_focus = YSS_dismiss_purge_directors
			}
			if = {
				limit = {
					has_completed_focus = YSS_dismiss_purge_directors
					NOT = { has_completed_focus = YSS_rehabilitate_wronged_cases }
				}
				country_event = { id = YSS_purge_aftermath.2 days = 70 }
			}
			else_if = {
				limit = { NOT = { has_completed_focus = YSS_redefine_party_spirit } }
				country_event = { id = YSS_purge_aftermath.2 }
			}
		}
	}
}
