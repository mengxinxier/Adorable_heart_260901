#同一片森林，同一个命运
country_event = {
    id = EWW.5
    title = EWW.5.t
    desc = EWW.5.d
    is_triggered_only = yes
    picture = GFX_event_wy
    immediate = {
    }
    option = {
        name = EWW.5.a
        ai_chance = {
            factor = 10
        }
        transfer_units_fraction= {
            target = EWW
            size = 1
            stockpile_ratio = 1
            keep_unit_leaders_trigger = {
                always = yes
            }
        }
        every_state = {
            limit = {
                is_core_of = ROOT
            }
            EWW = {
                add_core_of = PREV
            }
        }
		every_character = {
			limit = { is_general_advisor = no }
			set_nationality = EWW
		}
        every_owned_state = {
            limit = {
                is_fully_controlled_by = ROOT
            }
            add_core_of = EWW
            transfer_state_to = EWW
        }
        add_to_array = { global.FourSeasons_member = ROOT }
        add_to_variable = { global.num_of_FourSeasons = 1 }
        ###  计算动态阵营buff（如果有）
        ELF_FourSeasons_modifier_effect = yes
        ###  刷新经济阵营GDP
        compute_global_gdp = yes
    }
    option = {
        name = EWW.5.b
        ai_chance = {
            factor = 0
        }
    }
}
