	desc = YSS_sanjieduan.19.d
	is_triggered_only = yes
	fire_only_once = yes

	option = {
		name = YSS_sanjieduan.19.a
		# 战争姬出任最高领袖，子意识形态军事社会主义
		if = {
			limit = { has_country_leader = { character = YSS_byner } }
			retire_country_leader = yes
		}
		promote_character = {
			character = YSS_3_1
			ideology = Military_socialism
		}

		if = {
			limit = { has_idea = YSS_deputy_hanamaru }
			remove_ideas = YSS_deputy_hanamaru
		}
		add_ideas = YSS_deputy_lingcan
	}
}
