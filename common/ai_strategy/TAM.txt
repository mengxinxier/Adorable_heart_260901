	    allowed = {
	        original_tag = TAM
	    }
	    enable = {
	        num_of_military_factories > 20
	        has_equipment = { train_equipment < 40 }
	    }
	    abort_when_not_enabled = yes

	    ai_strategy = {
	        type = equipment_production_min_factories

	        value = 3
	    }
