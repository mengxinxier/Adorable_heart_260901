# 决议国家目标大洲筛选器（玩家可一键折叠/按大洲切换；AI不使用涉及外国地块/教团/强制附庸的决议）
# paq_target_filter: 0=已折叠, 1=南美, 2=北美, 3=欧洲, 4=亚洲, 5=非洲/大洋洲/中东
paq_target_matches_filter = {
	ROOT = { is_ai = no }
	OR = {
		AND = {
			ROOT = { check_variable = { paq_target_filter = 1 } }
			capital_scope = { is_on_continent = south_america }
		}
		AND = {
			ROOT = { check_variable = { paq_target_filter = 2 } }
			capital_scope = { is_on_continent = north_america }
		}
		AND = {
			ROOT = { check_variable = { paq_target_filter = 3 } }
			capital_scope = { is_on_continent = europe }
		}
		AND = {
			ROOT = { check_variable = { paq_target_filter = 4 } }
			capital_scope = { is_on_continent = asia }
		}
		AND = {
			ROOT = { check_variable = { paq_target_filter = 5 } }
			capital_scope = {
				OR = {
					is_on_continent = africa
					is_on_continent = australia
					is_on_continent = middle_east
				}
			}
		}
	}
}
