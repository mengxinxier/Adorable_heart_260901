#scripted trigger for RAJ - GOE


is_raj_or_raj_inheritor = {
	OR = {
		original_tag = RAJ
		original_tag = PAK
	}
}

is_raj_releasable = {
	OR = {
		original_tag = PAK
		is_in_array = { #mostly as failsafe
			array = RAJ.raj_releasables
			value = THIS
		}
	} 
}
