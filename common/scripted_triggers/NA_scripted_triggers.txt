# State scope; PREV must be the integrating country (e.g. ROOT in annexation events).
NA_is_same_language_state_as_PREV = {
	custom_trigger_tooltip = {
		tooltip = NA_is_same_language_state_tt
		OR = {
			AND = {
				OR = {
					has_state_flag = NAE
					AND = {
						has_state_flag = NA_ENG_AREA
						NOT = {
							has_state_flag = NAF
							has_state_flag = NAC
							has_state_flag = NAR
						}
					}
				}
				PREV = {
					OR = {
						tag = USA
						tag = CAN
					}
				}
			}
			AND = {
				has_state_flag = NAF
				PREV = {
					OR = {
						tag = NFA
						tag = FRG
					}
				}
			}
			AND = {
				has_state_flag = NAC
				PREV = {
					tag = NGD
				}
			}
			AND = {
				has_state_flag = NAR
				PREV = {
					tag = ASK
				}
			}
		}
	}
}
