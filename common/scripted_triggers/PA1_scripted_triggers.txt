Friendship_mutual_aid_faction = {
    clear_array = global.Friendship_mutual_aid_member
    set_variable = {global.num_of_Friendship_mutual_aid = 0}
    every_country = {
        limit = {
            check_variable = {THIS.economy_faction_var = 16}
            add_to_array = {global.Friendship_mutual_aid_member = THIS}
            add_to_variable = {global.num_of_Friendship_mutual_aid = 1}
        }
    }
    compute_global_gdp = yes
}
