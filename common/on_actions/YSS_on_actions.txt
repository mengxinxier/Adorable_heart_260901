on_actions = {
	# === YSS 内战、统一、非洲解放与自动国策推进 ===
	on_startup = {
		effect = {
			1410 = { set_state_flag = has_city_story } # 亚的斯亚贝巴城市故事
		}
	}

	on_capitulation_immediate = {
		effect = {
			if = {
				limit = {
					FROM = { tag = YSS }
					ROOT = { tag = ETH }
				}
				YSS = {
					white_peace = {
						tag = ETH
						message = YSS_win_ETH
					}
					annex_country = { target = ETH transfer_troops = no }
				}
			}
			if = {
				limit = {
					FROM = { tag = YSS }
					ROOT = { tag = AOI }
				}
				YSS = {
					white_peace = {
						tag = AOI
						message = YSS_win_AOI
					}
					annex_country = { target = AOI transfer_troops = no }
				}
			}
		}
	}

	# ── YSS 每日检查：大清洗善后自动推进 + 崩溃/内战胜利事件触发 ──
	on_daily_YSS = {
		effect = {
			if = {
				limit = {
					has_stability < 0.04
					NOT = { has_country_flag = YSS_2_triggered }
				}
				set_country_flag = YSS_2_triggered
				country_event = YSS.2
			}
			if = {
				limit = {
					has_country_flag = YSS_the_civil_war_of_YSS_launched
					NOT = { has_country_flag = YSS_win_war }
					NOT = { has_country_flag = YSS_90_triggered }
					NOT = { has_war_with = ETH }
					NOT = { has_war_with = AOI }
				}
				set_country_flag = YSS_90_triggered
				country_event = YSS.90
			}
			if = {
				limit = {
					has_country_flag = YSS_daqingxi_jiaoting
					NOT = { has_completed_focus = YSS_redefine_party_spirit }
				}
				# YSS_emergency_halt_to_purge（入口，需要 YSS_daqingxi_jiaoting flag）
				if = {
					limit = {
						NOT = { has_completed_focus = YSS_emergency_halt_to_purge }
					}
					if = {
						limit = { NOT = { has_variable = YSS_af_30 } }
						set_variable = { YSS_af_30 = 1 }
					}
					subtract_from_variable = { YSS_af_30 = 1 }
					if = {
						limit = { check_variable = { YSS_af_30 < 1 } }
						complete_national_focus = YSS_emergency_halt_to_purge
						clear_variable = YSS_af_30
					}
				}
				# YSS_dismiss_purge_directors（前置 YSS_emergency_halt_to_purge）
				else_if = {
					limit = {
						has_completed_focus = YSS_emergency_halt_to_purge
						NOT = { has_completed_focus = YSS_dismiss_purge_directors }
					}
					if = {
						limit = { NOT = { has_variable = YSS_af_31 } }
						set_variable = { YSS_af_31 = 70 }
					}
					subtract_from_variable = { YSS_af_31 = 1 }
					if = {
						limit = { check_variable = { YSS_af_31 < 1 } }
						complete_national_focus = YSS_dismiss_purge_directors
						clear_variable = YSS_af_31
					}
				}
				# YSS_rehabilitate_wronged_cases（前置 YSS_dismiss_purge_directors）
				else_if = {
					limit = {
						has_completed_focus = YSS_dismiss_purge_directors
						NOT = { has_completed_focus = YSS_rehabilitate_wronged_cases }
					}
					if = {
						limit = { NOT = { has_variable = YSS_af_32 } }
						set_variable = { YSS_af_32 = 70 }
					}
					subtract_from_variable = { YSS_af_32 = 1 }
					if = {
						limit = { check_variable = { YSS_af_32 < 1 } }
						complete_national_focus = YSS_rehabilitate_wronged_cases
						clear_variable = YSS_af_32
					}
				}
				# YSS_reorganize_the_politburo（前置 YSS_rehabilitate_wronged_cases）
				else_if = {
					limit = {
						has_completed_focus = YSS_rehabilitate_wronged_cases
						NOT = { has_completed_focus = YSS_reorganize_the_politburo }
					}
					if = {
						limit = { NOT = { has_variable = YSS_af_43 } }
						set_variable = { YSS_af_43 = 70 }
					}
					subtract_from_variable = { YSS_af_43 = 1 }
					if = {
						limit = { check_variable = { YSS_af_43 < 1 } }
						complete_national_focus = YSS_reorganize_the_politburo
						clear_variable = YSS_af_43
					}
				}
				# YSS_convene_politburo_session（前置 YSS_reorganize_the_politburo）
				else_if = {
					limit = {
						has_completed_focus = YSS_reorganize_the_politburo
						NOT = { has_completed_focus = YSS_convene_politburo_session }
					}
					if = {
						limit = { NOT = { has_variable = YSS_af_44 } }
						set_variable = { YSS_af_44 = 70 }
					}
					subtract_from_variable = { YSS_af_44 = 1 }
					if = {
						limit = { check_variable = { YSS_af_44 < 1 } }
						complete_national_focus = YSS_convene_politburo_session
						clear_variable = YSS_af_44
					}
				}
				# YSS_redefine_party_spirit（前置 YSS_convene_politburo_session）
				else_if = {
					limit = {
						has_completed_focus = YSS_convene_politburo_session
						NOT = { has_completed_focus = YSS_redefine_party_spirit }
					}
					if = {
						limit = { NOT = { has_variable = YSS_af_45 } }
						set_variable = { YSS_af_45 = 70 }
					}
					subtract_from_variable = { YSS_af_45 = 1 }
					if = {
						limit = { check_variable = { YSS_af_45 < 1 } }
						complete_national_focus = YSS_redefine_party_spirit
						clear_variable = YSS_af_45
					}
				}
			}
		}
	}

	# ── YSS 非洲革命内战胜利结算 ──
	on_civil_war_end = {
		effect = {
			if = {
				limit = {
					ROOT = {
						has_country_flag = YSS_launch_the_civil_war
					}
				}
				set_country_flag = YSS_revoluntion_victory
				clear_variable = ROOT.YSS_officer_factor
				remove_dynamic_modifier = { modifier = YSS_officer_support_revoluntion }

				if = {
					limit = {
						OR = {
							has_country_flag = YSS_have_volunteer_to_this_country
							has_country_flag = YSS_we_prepare_to_send_volunteer
						}
					}
					clr_country_flag = YSS_have_volunteer_to_this_country
					clr_country_flag = YSS_we_prepare_to_send_volunteer
					add_to_variable = { YSS.YSS_volunteer_to_other_country_send_volunteer_size = -1 }
				}

				# 战后归还：清除宗主标记、归还外派将领、返还借出军工
				random_country = {
					limit = {
						meta_trigger = {
							text = {
								has_country_flag = YSS_the_lord_of_[country_tag]
							}
							country_tag = "[ROOT.GetTag]"
						}
					}
					meta_effect = {
						text = {
							clr_country_flag = YSS_the_lord_of_[country_tag]
						}
						country_tag = "[ROOT.GetTag]"
					}
				}

				meta_effect = {
					text = {
						every_army_leader = {
							limit = { has_character_flag = YSS_has_mission_to_aboard }
							clr_character_flag = YSS_has_mission_to_aboard
							clr_character_flag = go_to_[country_tag]
							set_nationality = YSS
						}
					}
					country_tag = "[ROOT.GetTag]"
				}

				meta_effect = {
					text = {
						set_temp_variable = { clear_num_of_factory = YSS.YSS_total_of_factory_[country_tag] }
						set_variable = { YSS.YSS_total_of_factory_[country_tag] = 0 }
					}
					country_tag = "[ROOT.GetTag]"
				}

				meta_effect = {
					text = {
						set_temp_variable = { clean_num_of_factory = clear_num_of_factory }
						YSS = {
							add_offsite_building = {
								type = arms_factory
								level = clean_num_of_factory
							}
						}
						multiply_temp_variable = { clean_num_of_factory = -1 }
						add_to_variable = { YSS.YSS_the_num_of_send_factories = clean_num_of_factory }
						add_offsite_building = {
							type = arms_factory
							level = clean_num_of_factory
						}
					}
					test_number = "[?temp_var:clear_num_of_factory]"
				}

				YSS = {
					add_to_faction = ROOT
				}
			}
		}
	}

	# === YSS 第一个三年计划月度/周度计算 ===
	on_monthly_YSS = {
		effect = {
			if = {
				limit = { has_country_flag = YSS_open_sannianjihua }
				set_variable = { YSS_gy_upkeep_soft = 0 }
			if = {
				limit = { num_of_factories > 149 }
				set_variable = { YSS_gy_upkeep_soft = 15 }
			}
			else_if = {
				limit = { num_of_factories > 139 }
				set_variable = { YSS_gy_upkeep_soft = 14 }
			}
			else_if = {
				limit = { num_of_factories > 129 }
				set_variable = { YSS_gy_upkeep_soft = 13 }
			}
			else_if = {
				limit = { num_of_factories > 119 }
				set_variable = { YSS_gy_upkeep_soft = 12 }
			}
			else_if = {
				limit = { num_of_factories > 109 }
				set_variable = { YSS_gy_upkeep_soft = 11 }
			}
			else_if = {
				limit = { num_of_factories > 99 }
				set_variable = { YSS_gy_upkeep_soft = 10 }
			}
			else_if = {
				limit = { num_of_factories > 89 }
				set_variable = { YSS_gy_upkeep_soft = 9 }
			}
			else_if = {
				limit = { num_of_factories > 79 }
				set_variable = { YSS_gy_upkeep_soft = 8 }
			}
			else_if = {
				limit = { num_of_factories > 69 }
				set_variable = { YSS_gy_upkeep_soft = 7 }
			}
			else_if = {
				limit = { num_of_factories > 59 }
				set_variable = { YSS_gy_upkeep_soft = 6 }
			}
			else_if = {
				limit = { num_of_factories > 49 }
				set_variable = { YSS_gy_upkeep_soft = 5 }
			}
			else_if = {
				limit = { num_of_factories > 39 }
				set_variable = { YSS_gy_upkeep_soft = 4 }
			}
			else_if = {
				limit = { num_of_factories > 29 }
				set_variable = { YSS_gy_upkeep_soft = 3 }
			}
			else_if = {
				limit = { num_of_factories > 19 }
				set_variable = { YSS_gy_upkeep_soft = 2 }
			}
			else_if = {
				limit = { num_of_factories > 9 }
				set_variable = { YSS_gy_upkeep_soft = 1 }
			}
				add_to_variable = { YSS_gy_point = { value = YSS_gy_output } }
				add_to_variable = { YSS_gy_point = { value = YSS_gy_upkeep_hard multiply = -1 } }
				add_to_variable = { YSS_gy_point = { value = YSS_gy_upkeep_soft multiply = -1 } }
				# 不再把点数夹到 0：透支要留给 on_weekly_YSS 判 (<0 时触发拆厂事件)
				set_variable = { YSS_gy_upkeep_total = 0 }
				add_to_variable = { YSS_gy_upkeep_total = { value = YSS_gy_upkeep_hard } }
				add_to_variable = { YSS_gy_upkeep_total = { value = YSS_gy_upkeep_soft } }
			}
		}
	}


	on_weekly_YSS = {
		effect = {
			if = {
				limit = { check_variable = { YSS_gy_point < 0 } }
				country_event = { id = YSS_sannianjihua.12 }
			}
			if = {
				limit = {
					has_completed_focus = YSS_recontact_cpsu
					SOV = { has_war_with = DVL }
				}
				SOV = { country_event = { id = YSS_sanjieduan.3 } }
			}
			if = {
				limit = {
					has_completed_focus = YSS_recontact_cpsu
					SOV = { has_war_with = GER }
				}
				SOV = { country_event = { id = YSS_sanjieduan.4 } }
			}
			if = {
				limit = {
					date > 1942.1.1
					has_completed_focus = YSS_join_the_comintern
					NOT = { country_exists = DVL }
					OR = {
						NOT = { country_exists = GER }
						AND = {
							country_exists = GER
							GER = {
								NOT = { has_country_leader = { character = GER_Adolf_lisha } }
								NOT = { has_country_leader = { character = GER_xier } }
							}
						}
					}
					NOT = { has_country_flag = YSS_mlhm_dead }
				}
				country_event = { id = YSS_sanjieduan.1 }
			}
			if = {
				limit = {
					date > 1945.1.1
					NOT = { has_country_flag = YSS_mlhm_dead }
				}
				country_event = { id = YSS_sanjieduan.6 }
			}
		}
	}
}
