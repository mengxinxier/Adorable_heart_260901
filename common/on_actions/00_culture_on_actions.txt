on_actions = {
	on_startup = {
		effect = {
			every_state = {
				limit = { has_variable = state_culture }
				apply_state_culture_modifier = yes
			}
			every_country = {
				limit = { has_variable = country_culture_group }
				calculate_controlled_culture_arrays = yes
			}
		}
	}
	on_state_control_changed = {
		effect = {
			FROM.FROM = {
				if = {
					limit = { has_variable = state_culture }
					apply_state_culture_modifier = yes
				}
			}
			if = {
				limit = { ROOT = { has_variable = country_culture_group } }
				ROOT = { calculate_controlled_culture_arrays = yes }
			}
			if = {
				limit = { FROM = { has_variable = country_culture_group } }
				FROM = { calculate_controlled_culture_arrays = yes }
			}
		}
	}
	on_monthly = {
		effect = {
			if = {
				limit = { has_variable = country_culture_group }
				calculate_controlled_culture_arrays = yes
			}
		}
	}
}
