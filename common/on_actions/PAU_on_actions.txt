#PAU全局逻辑
on_actions = {
    on_war = {
        effect = {
            PAU_try_fire_grail_war_news = yes
            PAU = { PAU_refresh_targeted_spirit_visibility = yes }
        }
    }
    on_war_relation_added = {
        effect = {
            if = {
                limit = {
                    NOT = { has_global_flag = PAU_grail_war_news_fired }
                    OR = {
                        tag = PAU
                        tag = ENG
                        FROM = { tag = PAU }
                        FROM = { tag = ENG }
                    }
                }
                PAU_try_fire_grail_war_news = yes
            }
        }
    }
    on_startup = {
        effect = {
            PAU = {
                add_to_variable = {
                    var = ubs_load
                    value = 100
                }

                    if = { limit = { NOT = { has_variable = PAU_mercenary_reputation } } set_variable = { PAU_mercenary_reputation = 50 } }  #雇佣兵信誉=初始50+合同事件增减
                    if = { limit = { NOT = { has_variable = PAU_grail_rations } } set_variable = { PAU_grail_rations = 100 } }  #圣杯流亡营口粮=初始100-分粮/贸易/换零件消耗
                    if = { limit = { NOT = { has_variable = wl_attack_day } } set_variable = { wl_attack_day = 7 } }
                    if = { limit = { NOT = { has_variable = PAU_mil_reform_progress_frame } } set_variable = { PAU_mil_reform_progress_frame = 0 } }
                    if = { limit = { NOT = { has_variable = PAU_eco_reform_progress_frame } } set_variable = { PAU_eco_reform_progress_frame = 0 } }
                    PAU_sync_baiye_gui_vars = yes
                    if = {
                        limit = { NOT = { has_country_flag = PAU_reform_reset_initialized } }
                        set_country_flag = PAU_reform_reset_initialized
                        set_variable = { PAU_reform_reset_count = 1 }
                    }
                }
            }
        }
    on_monthly = { 
        effect = {
            if = {
                limit = { original_tag = PAU }
                if = {
                    limit = {
                        check_variable = { ubs_save < 0 }  #王室欠饷债务余额=每月把负值折算进ubs_load
                    }
                    subtract_from_variable = { ubs_load = ubs_save }
                    clamp_variable = {
                        var = ubs_save
                        min = 0
                    }
                }
                if = {
                    limit = {
                        has_country_flag = PAU_safe_harbor_unlocked  #避风港机制解锁标记
                    }
                    PAU_refresh_harbor_spillover_states = yes
                }
                if = { limit = { check_variable = { PAU_axia_mil > 0 } } set_country_flag = PAU_army_reform_axia_unlocked }  #阿夏军事改革点=军改事件+20，改革决议-20
                if = { limit = { check_variable = { PAU_xfxy_mil > 0 } } set_country_flag = PAU_army_reform_xfxy_unlocked }  #晓枫雪伊军事改革点=军改事件+20，改革决议-20
                if = { limit = { check_variable = { PAU_xiaoyao_mil > 0 } } set_country_flag = PAU_army_reform_xiaoyao_unlocked }  #小遥军事改革点=军改事件+20，改革决议-20
                if = { limit = { check_variable = { PAU_axia_eco > 0 } } set_country_flag = PAU_economy_reform_axia_unlocked }  #阿夏经济改革点=经改事件+20，改革决议-20
                if = { limit = { check_variable = { PAU_xfxy_eco > 0 } } set_country_flag = PAU_economy_reform_xfxy_unlocked }  #晓枫雪伊经济改革点=经改事件+20，改革决议-20
                if = { limit = { check_variable = { PAU_xiaoyao_eco > 0 } } set_country_flag = PAU_economy_reform_xiaoyao_unlocked }  #小遥经济改革点=经改事件+20，改革决议-20
                PAU_refresh_targeted_spirit_visibility = yes
                PAU_refresh_south_america_members = yes
                PAU_refresh_south_america_development = yes
                PAU_refresh_state_council = yes
                PAU_refresh_mil_reform_tiers = yes
                PAU_refresh_eco_reform_tiers = yes
                if = {
                    limit = {
                        NOT = { has_country_flag = PAU_7005_fired }
                        OR = {
                            has_country_flag = PAU_pending_emma_rescue
                            has_country_flag = PAU_undead_defeated
                            has_completed_focus = PAU_ddjr
                        }
                    }
                    PAU_check_baiye_story_progression = yes
                }
            }
            if = {
                limit = {
                    any_country = {
                        has_defensive_war_with = JAP
                        not = {
                            has_country_flag = pau_jap_defence_flag
                        }
                    }
                }
                every_country = {
                    limit = {
                        not = {
                            tag = PAU
                            has_country_flag = pau_jap_defence_flag
                        }
                        has_defensive_war_with = JAP
                    }
                    set_country_flag = pau_jap_defence_flag
                }
            }
        }
    }
    on_capitulation = {
        effect = {
            if = {
                limit = {
                    ROOT = {
                        has_country_flag = PAU_bop_loan_active  #BOP雇佣贷款协议生效标记
                    }
                }
                every_country = {
                    limit = {
                        has_country_flag = PAU_bop_loan_active
                    }
                    every_country = {
                        limit = {
                            has_war_with = PREV
                        }
                        set_country_flag = PAU_guzhu_cap  #雇主投降标记
                    }
                }
            }
        }
    }
	on_state_control_changed = {
		effect = {
				if = {
					limit = {
						FROM.FROM = {
							state = 282
						}
						PAU = {
							controls_province = 1182
						}
					}
                    PAU = {
                        country_event = {
                            id = PAU.43
                            hours = 6
                        }
                    }
            }
				if = {
					limit = {
						ROOT = {
							OR = {
								tag = PAU
								is_in_faction_with = PAU
							}
						}
						FROM = { tag = ZDW }
						FROM.FROM = {
							has_state_flag = is_lldy_state_flag
							NOT = { has_state_flag = PAU_zdw_retaken_bonus }
						}
					}
					FROM.FROM = { set_state_flag = PAU_zdw_retaken_bonus }
					PAU = {
						army_experience = 5
						add_command_power = 10
						add_manpower = 1000
						if = {
							limit = { has_war_support < 0.85 }
							add_war_support = 0.01
						}
						else = {
							add_political_power = 10
						}
					}
					ZDW = { add_war_support = -0.01 }
				}
        }
    }
	#PAU每日逻辑
	on_daily = {
		effect = {
			if = {
				limit = { original_tag = PAU }
				PAU_daily_reform_tick = yes
				if = {
					limit = {
						NOT = { has_global_flag = PAU_grail_war_news_fired }
						has_war_with = ENG
					}
					PAU_try_fire_grail_war_news = yes
				}
			}
		}
	}
	on_declare_war = {
		effect = {
			if = {
				limit = {
					NOT = { has_global_flag = PAU_grail_war_news_fired }
					OR = {
						tag = PAU
						tag = ENG
						FROM = { tag = PAU }
						FROM = { tag = ENG }
					}
				}
				PAU_try_fire_grail_war_news = yes
			}
			if = {
				limit = { PAU = { has_country_flag = PAU_guyong_tracking_active } }
				if = {
					limit = {
						ROOT = {
							OR = {
								tag = PAU
								is_in_faction_with = PAU
								is_subject_of = PAU
							}
						}
						FROM = {
							NOT = { has_country_flag = PAU_will_guyonged_touxiang_flag }
							NOT = { is_in_array = { array = PAU.PAU_contract_enemies value = THIS } }
							NOT = {
								any_allied_country = {
									OR = {
										has_country_flag = PAU_will_guyonged_touxiang_flag
										is_in_array = { array = PAU.PAU_contract_enemies value = THIS }
									}
								}
							}
						}
					}
					ROOT = {
						if = {
							limit = { NOT = { is_in_array = { array = PAU_pre_contract_enemies value = FROM } } }
							add_to_array = { PAU_pre_contract_enemies = FROM }
						}
					}
				}
				if = {
					limit = {
						FROM = {
							OR = {
								tag = PAU
								is_in_faction_with = PAU
								is_subject_of = PAU
							}
						}
						ROOT = {
							NOT = { has_country_flag = PAU_will_guyonged_touxiang_flag }
							NOT = { is_in_array = { array = PAU.PAU_contract_enemies value = THIS } }
							NOT = {
								any_allied_country = {
									OR = {
										has_country_flag = PAU_will_guyonged_touxiang_flag
										is_in_array = { array = PAU.PAU_contract_enemies value = THIS }
									}
								}
							}
						}
					}
					FROM = {
						if = {
							limit = { NOT = { is_in_array = { array = PAU_pre_contract_enemies value = ROOT } } }
							add_to_array = { PAU_pre_contract_enemies = ROOT }
						}
					}
				}
			}
		}
	}
	on_join_faction = {
		effect = {
			PAU_sync_active_ally_pre_contract_wars = yes
		}
	}
	on_offer_join_faction = {
		effect = {
			PAU_sync_active_ally_pre_contract_wars = yes
		}
	}
	on_puppet = {
		effect = {
			PAU = { PAU_refresh_targeted_spirit_visibility = yes }
			PAU_sync_active_ally_pre_contract_wars = yes
		}
	}
	on_annex = {
		effect = {
			PAU = { PAU_refresh_targeted_spirit_visibility = yes }
		}
	}
	on_peaceconference_ended = {
		effect = {
			PAU = { PAU_refresh_targeted_spirit_visibility = yes }
		}
	}
}
