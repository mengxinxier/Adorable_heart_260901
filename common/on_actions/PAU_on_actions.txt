#PAU全局逻辑
on_actions = {
    on_startup = {
        effect = {
            every_state = {
                limit = {
                    or = {
                        is_core_of = PAM
                        is_core_of = CHL
                        is_core_of = PAU
                        is_core_of = LHS
                    }
                }
                set_state_flag = is_lldy_state_flag  #莉兰迪亚核心州标记
            }
            every_country = {
                if = {
                    limit = {
                        tag = PAU
                    }
                    add_to_variable = {
                        var = ubs_load
                        value = 100
                    }

                    country_event = {
                        id = PAU.11111
                    }

                    if = { limit = { NOT = { has_variable = PAU_mercenary_reputation } } set_variable = { PAU_mercenary_reputation = 50 } }  #雇佣兵信誉=初始50+合同事件增减
                    if = { limit = { NOT = { has_variable = PAU_grail_rations } } set_variable = { PAU_grail_rations = 100 } }  #圣杯流亡营口粮=初始100-分粮/贸易/换零件消耗
                    if = { limit = { NOT = { has_variable = wl_attack_day } } set_variable = { wl_attack_day = 7 } }
                }
            }
        }
    }
    on_monthly = {
        effect = {
            if = {
                limit = { original_tag = PAU }
                if = {
                    limit = {
                        check_variable = { ubs_save < 0 }  #王室欠饷债务余额=每月把负值折算进ubs_load
                    }
                    subtract_from_variable = { ubs_load = ubs_save }
                    clamp_variable = {
                        var = ubs_save
                        min = 0
                    }
                }
                if = {
                    limit = {
                        has_country_flag = PAU_safe_harbor_unlocked  #避风港机制解锁标记
                    }
                    PAU_refresh_harbor_spillover_states = yes
                }
                if = { limit = { check_variable = { PAU_axia_mil > 0 } } set_country_flag = PAU_army_reform_axia_unlocked }  #阿夏军事改革点=军改国策+10，改革决议-10
                if = { limit = { check_variable = { PAU_xfxy_mil > 0 } } set_country_flag = PAU_army_reform_xfxy_unlocked }  #晓枫雪伊军事改革点=军改国策+10，改革决议-10
                if = { limit = { check_variable = { PAU_xiaoyao_mil > 0 } } set_country_flag = PAU_army_reform_xiaoyao_unlocked }  #小遥军事改革点=军改国策+10，改革决议-10
                if = { limit = { check_variable = { PAU_axia_eco > 0 } } set_country_flag = PAU_economy_reform_axia_unlocked }  #阿夏经济改革点=经改国策+10，改革决议-10
                if = { limit = { check_variable = { PAU_xfxy_eco > 0 } } set_country_flag = PAU_economy_reform_xfxy_unlocked }  #晓枫雪伊经济改革点=经改国策+10，改革决议-10
                if = { limit = { check_variable = { PAU_xiaoyao_eco > 0 } } set_country_flag = PAU_economy_reform_xiaoyao_unlocked }  #小遥经济改革点=经改国策+10，改革决议-10
                PAU_refresh_south_america_members = yes
                PAU_refresh_south_america_development = yes
                PAU_refresh_state_council = yes
                PAU_refresh_mil_reform_tiers = yes
                PAU_refresh_eco_reform_tiers = yes
            }
            if = {
                limit = {
                    any_country = {
                        has_defensive_war_with = JAP
                        not = {
                            has_country_flag = pau_jap_defence_flag
                        }
                    }
                }
                every_country = {
                    limit = {
                        not = {
                            tag = PAU
                            has_country_flag = pau_jap_defence_flag
                        }
                        has_defensive_war_with = JAP
                    }
                    set_country_flag = pau_jap_defence_flag
                }
            }
        }
    }
    on_capitulation = {
        effect = {
            if = {
                limit = {
                    ROOT = {
                        has_country_flag = PAU_bop_loan_active  #BOP雇佣贷款协议生效标记
                    }
                }
                every_country = {
                    limit = {
                        has_country_flag = PAU_bop_loan_active
                    }
                    every_country = {
                        limit = {
                            has_war_with = PREV
                        }
                        set_country_flag = PAU_guzhu_cap  #雇主投降标记
                    }
                }
            }
        }
    }
	on_state_control_changed = {
		effect = {
			if = {
				if = {
					limit = {
						FROM.FROM = {
							state = 282
						}
						PAU = {
							controls_province = 1182
						}
					}
                    PAU = {
                        country_event = {
                            id = PAU.43
                            hours = 6
                        }
                    }
				}
            }
        }
    }
}
