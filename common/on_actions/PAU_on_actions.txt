# PAU 每月事件：亡灵压力结算、自动任务重挂、变量兜底与夜莺之国结局检查。
on_actions = {
    on_startup = {
        effect = {
            every_state = {
                limit = {
                    or = {
                        is_core_of = PAM
                        is_core_of = CHL
                        is_core_of = PAU
                        is_core_of = LHS
                    }
                }
                set_state_flag = is_lldy_state_flag
            }
            every_country = {
                if = {
                    limit = {
                        tag = PAU
                    }
                    add_to_variable = {
                        var = ubs_load
                        value = 100
                    }

                    # 旧存档兜底：只补非零默认值
                    if = { limit = { NOT = { has_variable = PAU_mercenary_reputation } } set_variable = { PAU_mercenary_reputation = 50 } }
                    if = { limit = { NOT = { has_variable = PAU_grail_rations } } set_variable = { PAU_grail_rations = 100 } }
                    if = { limit = { NOT = { has_variable = PAU_grail_parts } } set_variable = { PAU_grail_parts = 100 } }
                    if = { limit = { NOT = { has_variable = wl_attack_day } } set_variable = { wl_attack_day = 7 } }
                    # 仍被 check_variable 依赖的状态机计数器
                    if = { limit = { NOT = { has_variable = PAU_ml_count_PAU_revolution_network } } set_variable = { PAU_ml_count_PAU_revolution_network = 0 } }
                    if = { limit = { NOT = { has_variable = PAU_ml_count_PAU_revolution_network_2 } } set_variable = { PAU_ml_count_PAU_revolution_network_2 = 0 } }
                    if = { limit = { NOT = { has_variable = PAU_ml_count_PAU_revolution_network_3 } } set_variable = { PAU_ml_count_PAU_revolution_network_3 = 0 } }
                    if = { limit = { NOT = { has_variable = PAU_ml_count_PAU_trade_network } } set_variable = { PAU_ml_count_PAU_trade_network = 0 } }
                    if = { limit = { NOT = { has_variable = PAU_ml_count_PAU_trade_network_2 } } set_variable = { PAU_ml_count_PAU_trade_network_2 = 0 } }
                    if = { limit = { NOT = { has_variable = PAU_ml_count_PAU_trade_network_3 } } set_variable = { PAU_ml_count_PAU_trade_network_3 = 0 } }
                    if = { limit = { NOT = { has_variable = PAU_ml_count_PAU_coastal_network } } set_variable = { PAU_ml_count_PAU_coastal_network = 0 } }
                    if = { limit = { NOT = { has_variable = PAU_ml_count_PAU_coastal_network_2 } } set_variable = { PAU_ml_count_PAU_coastal_network_2 = 0 } }
                    if = { limit = { NOT = { has_variable = PAU_ml_count_PAU_coastal_network_3 } } set_variable = { PAU_ml_count_PAU_coastal_network_3 = 0 } }
                    if = { limit = { NOT = { has_variable = PAU_ml_count_PAU_undead_pressure } } set_variable = { PAU_ml_count_PAU_undead_pressure = 0 } }
                }
            }
        }
    }
    on_monthly = {
        effect = {
            if = {
                limit = { original_tag = PAU }
                if = {
                    limit = {
                        check_variable = { ubs_save < 0 }
                    }
                    subtract_from_variable = { ubs_load = ubs_save }
                    clamp_variable = {
                        var = ubs_save
                        min = 0
                    }
                }
            if = {
                limit = {
                    has_country_flag = PAU_safe_harbor_unlocked
                }
                PAU_refresh_harbor_spillover_states = yes
            }
            if = {
                limit = { has_country_flag = PAU_undead_war }
                PAU_refresh_undead_pressure = yes
            }
            if = { limit = { check_variable = { PAU_axia_mil > 0 } } set_country_flag = PAU_army_reform_axia_unlocked } # 阿夏系军事改革点：国策+10，改革决议-10
            if = { limit = { check_variable = { PAU_xfxy_mil > 0 } } set_country_flag = PAU_army_reform_xfxy_unlocked } # 晓枫雪伊系军事改革点：国策+10，改革决议-10
            if = { limit = { check_variable = { PAU_xiaoyao_mil > 0 } } set_country_flag = PAU_army_reform_xiaoyao_unlocked } # 小遥系军事改革点：国策+10，改革决议-10
            if = { limit = { check_variable = { PAU_axia_eco > 0 } } set_country_flag = PAU_economy_reform_axia_unlocked } # 阿夏系经济改革点：国策+10，改革决议-10
            if = { limit = { check_variable = { PAU_xfxy_eco > 0 } } set_country_flag = PAU_economy_reform_xfxy_unlocked } # 晓枫雪伊系经济改革点：国策+10，改革决议-10
            if = { limit = { check_variable = { PAU_xiaoyao_eco > 0 } } set_country_flag = PAU_economy_reform_xiaoyao_unlocked } # 小遥系经济改革点：国策+10，改革决议-10
            PAU_refresh_late_networks = yes
            PAU_refresh_south_america_members = yes
            PAU_refresh_south_america_development = yes
            PAU_refresh_joint_staff_college = yes
            PAU_refresh_state_council = yes
            PAU_refresh_mil_reform_tiers = yes
            PAU_refresh_eco_reform_tiers = yes
            }
            }
        }
    on_capitulation = {
        effect = {
            if = {
                limit = {
                    ROOT = {
                        has_country_flag = PAU_bop_loan_active #同意的flag
                    }
                }
                every_country = {
                    limit = {
                        has_country_flag = PAU_bop_loan_active #同意的flag
                    }
                    every_country = {
                        limit = {
                            has_war_with = PREV
                        }
                        set_country_flag = PAU_guzhu_cap
                    }
                }
            }
        }
    }
}
