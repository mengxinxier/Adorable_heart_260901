on_actions = {
	on_new_term_election = {
		random_events = {
			100 = BAL_ele.12
		}
	}
	on_civil_war_end_before_annexation = {
    	effect = {
			if = {
				limit = { has_global_flag = BAL_civil_war_has_begun_flag }
				FROM = {
					if = {
						limit = { original_tag = BAL }
						every_character = {
							limit = { is_general_advisor = no }
							set_nationality = AHM
						}
					}
				}
			}
		}
	}
	on_annex = {
		effect = {
			if = {
				limit = { has_global_flag = BAL_civil_war_has_begun_flag }
				if = {
					limit = { FROM = { original_tag = BAL } }
					ROOT = { country_event = BAL_cw.02000 }
				}
			}
		}
	}
}
