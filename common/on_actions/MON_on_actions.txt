on_actions = {
	on_state_control_changed = {
		effect = {
			if = {
				limit = {
					FROM.FROM = { state = 716 }
					FROM = {
						has_war_with = ROOT
						original_tag = MAN
					}
					ROOT = {
						has_war_with = FROM
						OR = {
						AND = {
						tag = MON
						has_completed_focus = MON_tuxifengtian
						}						
						AND = {
						tag = MON
						has_completed_focus = MON_tuxifengtian2
						}
						}
						has_idea = MON_tuxi_idea
					}
				}
			MAN = {
				transfer_navy = {
					target = MON
				}
					}
				ROOT = {
					white_peace = {
						tag = FROM
					}
					annex_country = { target = FROM }
					country_event = mon_politics.2
				}
			}			
			if = {
				limit = {
					FROM.FROM = { state = 608 }
					FROM = {
						has_war_with = ROOT
						original_tag = MON
					}
					ROOT = {
						has_war_with = FROM
						original_tag = GXC
						is_subject = no
					}
				}
			MON = {
				transfer_navy = {
					target = GXC
				}
					}
				ROOT = {
					white_peace = {
						tag = FROM
					}
					annex_country = { target = FROM }
					country_event = mon_politics.2
				}
			}
		}
	}
	on_army_leader_daily = {
		effect = {
			if = {
				limit = {
					skill > 2
					has_character_flag = MON_character_white_new_officer_flag
				}
				unit_leader_event = {
					id = mon_flavor.45
				}
			}
		}
	}
	on_war = {
		effect = {
			if = {
				limit = {
					original_tag = MON
					has_war = yes
					is_subject = no
					NOT = {
						has_idea = MON_dayuanzheng_idea
					}
					has_country_leader = {
						character = MON_ai
						ruling_only = yes
					}
					has_country_flag = MON_dayuanzheng_bop_flag
				}
				country_event = mon_flavor.47
			}
			}
	}	
	on_peace = {
		effect = {
			if = {
				limit = {
					original_tag = MON
					has_war = no
					is_subject = no
					NOT = {
						has_idea = MON_junxinhuansan_idea
					}
					has_country_leader = {
						character = MON_ai
						ruling_only = yes
					}
					has_country_flag = MON_dayuanzheng_bop_flag
				}
				country_event = mon_flavor.48
			}
			}
	}
}
