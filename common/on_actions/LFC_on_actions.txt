on_actions = {

	on_startup = {
		effect = { #事件
			#大萧条是2月21日，第51天
			LFC = {
			country_event = {  #经济危机波及我国
				id = LFC_event.101
				days = 54
			}
			}


		}

		effect = { #地区修正
			SAJ = {
				every_state = {
					limit = {
						is_core_of = SAJ  
					}

					add_dynamic_modifier = {
						modifier = LFC_state_dynamic_modifier_Rainforest_wktdq
					}	
				}
				1202 = {
					set_variable = { VAR_LFC_Rainforest_yjktcs = 1 }
				}
				1203 = {
					set_variable = { VAR_LFC_Rainforest_yjktcs = 1 }
				}
				1204 = {
					set_variable = { VAR_LFC_Rainforest_yjktcs = 1 }
				}
				1205 = {
					set_variable = { VAR_LFC_Rainforest_yjktcs = 1 }
				}
				1206 = {
					set_variable = { VAR_LFC_Rainforest_yjktcs = 1 }
				}
				1207 = {
					set_variable = { VAR_LFC_Rainforest_yjktcs = 1 }
				}
				1208 = {
					set_variable = { VAR_LFC_Rainforest_yjktcs = 1 }
				}
				504 = {
					set_variable = { VAR_LFC_Rainforest_yjktcs = 1 }
				}
				505 = {
					set_variable = { VAR_LFC_Rainforest_yjktcs = 1 }
				}
				280 = {
					set_variable = { VAR_LFC_Rainforest_yjktcs = 1 }
				}
			}
			LFC = {
				1202 = {
					add_dynamic_modifier = {
						modifier = LFC_state_dynamic_modifier_Rainforest_yzjqs
					}
				}
			}
		}
		
	}


	# 露斯契亚边疆：升级前已开着地图的旧存档在下一日补齐帧号。
	on_daily_LFC = {
		effect = {
			if = {
				limit = { has_country_flag = FLAG_LFC_frontier_open NOT = { has_country_flag = FLAG_LFC_frontier_frames_ready } }
				LFC_frontier_refresh_map_frames = yes
			}
		}
	}

	# 新控制者为露斯契亚时从该州当前修正重算帧号。
	on_state_control_changed = {
		effect = {
			if = {
				limit = {
					tag = LFC
					FROM.FROM = { OR = { state = 1202 state = 1203 state = 1204 state = 1205 state = 1206 state = 1207 state = 1208 state = 280 state = 504 state = 505 } }
				}
				FROM.FROM = { LFC_frontier_refresh_state_frame = yes }
			}
		}
	}

	on_monthly_LFC = {
		effect = { #每月遗物增长
			add_to_variable = {
				LFC_VAR_Rainforest_product = LFC_VAR_Rainforest_product_monthly
			}
		}
		effect = { #每月调度许可增长
			LFC_Preset_monthly_license_calculation = yes
			
			add_to_variable = {
				LFC_VAR_Empire_license = LFC_VAR_Empire_license_monthly_end
			}
		}
	}	
}
