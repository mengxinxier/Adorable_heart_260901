on_actions = {
	on_startup = {
		effect = {
			KNM = {
            	country_event = { #春天开始
            	    id = KNM_spring.10001
            	    days = 59
            	}
            	country_event = { #春天结束
            	    id = KNM_spring.10002
            	    days = 151
            	}
            	country_event = { #春天开始
            	    id = KNM_spring.10003
            	    days = 243
            	}
            	country_event = { #春天结束
            	    id = KNM_spring.10004
            	    days = 334
            	}
			}
		}
	}
    on_declare_war = {
        effect = {
            #判定参战双方是否为DVL与KNM
            if = {
                limit = {
                    OR = {
                        AND = {
                            ROOT = { tag = DVL }
                            FROM = { tag = KNM }
                        }
                        AND = {
                            ROOT = { tag = KNM }
                            FROM = { tag = DVL }
                        }
                    }
                    #确认幻想乡已同意
                    FRA = {
						has_country_flag = invitedfra_yes
                    }
                }
                #让幻想乡一天后收到参战事件
                FRA = {
					declare_war_on = {
						target = DVL
						type = annex_everything
					}
                }
            }
        }
    }
}
