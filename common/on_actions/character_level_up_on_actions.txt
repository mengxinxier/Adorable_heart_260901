on_actions = {
	on_unit_leader_level_up = {
		effect = {
			#人物升级提示
			if = {
				limit = {
					from = { is_ai = no }
				}
				character = {
					save_global_event_target_as = on_unit_leader_level_up_prompt_alerts
			
					add_to_array = { from.adorable_heart_prompt_alerts = 1 }
					add_to_array = { from.adorable_heart_prompt_alert_other_var1 = skill_level }
					add_to_array = { from.adorable_heart_prompt_alert_other_var2 = attack_level }
					add_to_array = { from.adorable_heart_prompt_alert_other_var3 = defense_level }
					if = {
						limit = { is_army_leader = yes }
						add_to_array = { from.adorable_heart_prompt_alert_other_var4 = planning_level }
						add_to_array = { from.adorable_heart_prompt_alert_other_var5 = logistics_level }
					}
					else = {
						add_to_array = { from.adorable_heart_prompt_alert_other_var4 = maneuvering_level }
						add_to_array = { from.adorable_heart_prompt_alert_other_var5 = coordination_level }
					}
					add_to_array = { from.adorable_heart_prompt_alert_flags = from }
					add_to_array = { from.adorable_heart_prompt_alert_flags2 = from }
					add_to_array = { from.adorable_heart_prompt_alerts_progressbars = -317 }
					set_variable_to_random = global.adorable_heart_alerts_dirty
				}
				from = {
					scoped_sound_effect = "diplomatic_notification"
					country_event = { id = alerts_mechanism.1 }
				}
			}
		}
	}
}
