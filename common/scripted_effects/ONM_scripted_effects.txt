ONM_total_state_population_k = {
	set_variable = {total_state_population_k = 0}
	every_owned_state = {
		add_to_variable = { ONM.total_state_population_k = state_population_k }
	}
	round_variable = total_state_population_k
}
ONM_Worker_calculate = {
	set_variable = { onm_total_Worker = 0 }
	#variable = { onm_proportion_of_workers = 0.6 }
	set_variable = { onm_proportion_of_workers_1 = onm_proportion_of_workers }
	multiply_variable = { onm_proportion_of_workers_1 = 100 }
	add_to_variable = { onm_total_Worker = total_state_population_k }
	multiply_variable = { onm_total_Worker = onm_proportion_of_workers }
	round_variable = onm_total_Worker
}
ONM_education_calculate = {
	#variable = { onm_proportion_of_education = 0.1 }
	set_variable = { onm_education_Worker = 0 }
	set_variable = { onm_education_Worker_1 = onm_proportion_of_education }
	multiply_variable = { onm_education_Worker_1 = 100 }
	add_to_variable = { onm_education_Worker = onm_total_Worker }
	multiply_variable = { onm_education_Worker = onm_proportion_of_education }
	round_variable = onm_education_Worker
}
ONM_Farm_worker_calculate = {
	#variable = { onm_farm_efficiency = 1 }
	#variable = { onm_Imported_food = 0 }
	set_variable = { ONM_Farm_worker = 1 }
	subtract_from_variable = { ONM_Farm_worker = onm_Imported_food }
	multiply_variable = { ONM_Farm_worker = total_state_population_k }
	divide_variable = { ONM_Farm_worker = onm_farm_efficiency }
	if = {
		limit = {
			check_variable = { ONM_Farm_worker > onm_total_Worker }
		}
		set_variable = { ONM_Farm_worker = onm_total_Worker }
	}
	round_variable = ONM_Farm_worker
}
ONM_industry_worker_calculate = {
	set_variable = { ONM_industry_worker = 0 }
	if = {
		limit = {
			not = {
				check_variable = { ONM_Farm_worker < onm_total_Worker }
			}
		}
	}
	else = {
		add_to_variable = { ONM_industry_worker = onm_total_Worker }
		subtract_from_variable = { ONM_industry_worker = ONM_Farm_worker }
		round_variable = ONM_industry_worker
		if = {
			limit = {
				check_variable = { ONM_industry_worker > onm_education_Worker }
			}
			set_variable = { ONM_industry_worker = onm_education_Worker }
		}
	}
}

					region = 167

					region = 215
				}

				not = { state = 326 }
				not = { state = 1052 }
			}
			set_variable = { ONM.onm_temp = state_population_k }
			multiply_variable = { ONM.onm_temp = ONM.onm_trial_ceremony_random }
			add_to_array = { 
				array = ONM.onm_trial_ceremony_num
				value = ONM.onm_temp
			}
			add_to_array = { 
				array = ONM.onm_trial_ceremony_state_id
				value = THIS.id
			}
		}
		clear_variable = ONM.onm_trial_ceremony_total_amount
		for_loop_effect = {
			end = ONM.onm_trial_ceremony_num^num
			add_to_variable = { ONM.onm_trial_ceremony_total_amount = ONM.onm_trial_ceremony_num^v }
