FRA_Calculate_local_ideology = {
	set_variable = { THIS.Total_local_ideology_var = 0 }
	#计算百分比
	add_to_variable = { THIS.Total_local_ideology_var = Local_Influence_of_LCR }
	add_to_variable = { THIS.Total_local_ideology_var = Local_Influence_of_PCG }
	add_to_variable = { THIS.Total_local_ideology_var = Local_Influence_of_PSG }
	add_to_variable = { THIS.Total_local_ideology_var = Local_Influence_of_PFG }
	add_to_variable = { THIS.Total_local_ideology_var = Local_Influence_of_IDU }
	add_to_variable = { THIS.Total_local_ideology_var = Local_Influence_of_GFN }
	add_to_variable = { THIS.Total_local_ideology_var = Local_Influence_of_MNR }
	divide_variable = { THIS.Local_Influence_of_LCR = THIS.Total_local_ideology_var }
	multiply_variable = { THIS.Local_Influence_of_LCR = 100 }
	divide_variable = { THIS.Local_Influence_of_PCG = THIS.Total_local_ideology_var }
	multiply_variable = { THIS.Local_Influence_of_PCG = 100 }
	divide_variable = { THIS.Local_Influence_of_PSG = THIS.Total_local_ideology_var }
	multiply_variable = { THIS.Local_Influence_of_PSG = 100 }
	divide_variable = { THIS.Local_Influence_of_PFG = THIS.Total_local_ideology_var }
	multiply_variable = { THIS.Local_Influence_of_PFG = 100 }
	divide_variable = { THIS.Local_Influence_of_IDU = THIS.Total_local_ideology_var }
	multiply_variable = { THIS.Local_Influence_of_IDU = 100 }
	divide_variable = { THIS.Local_Influence_of_GFN = THIS.Total_local_ideology_var }
	multiply_variable = { THIS.Local_Influence_of_GFN = 100 }
	divide_variable = { THIS.Local_Influence_of_MNR = THIS.Total_local_ideology_var }
	multiply_variable = { THIS.Local_Influence_of_MNR = 100 }

	if = {
		limit = {
			NOT = {
				or = {
					check_variable = { Local_Influence_of_LCR < Local_Influence_of_PCG }

					check_variable = { Local_Influence_of_LCR < Local_Influence_of_PFG }
					check_variable = { Local_Influence_of_LCR < Local_Influence_of_IDU }
					check_variable = { Local_Influence_of_LCR < Local_Influence_of_GFN }
					check_variable = { Local_Influence_of_LCR < Local_Influence_of_MNR }
				}
			}
		}
		set_variable = { FRA_election_states_var = 2 }
	}
	else_if = {
		limit = {
			NOT = {
				or = {
					check_variable = { Local_Influence_of_PCG < Local_Influence_of_LCR }
					check_variable = { Local_Influence_of_PCG < Local_Influence_of_PSG }
					check_variable = { Local_Influence_of_PCG < Local_Influence_of_PFG }
					check_variable = { Local_Influence_of_PCG < Local_Influence_of_IDU }
					check_variable = { Local_Influence_of_PCG < Local_Influence_of_GFN }
					check_variable = { Local_Influence_of_PCG < Local_Influence_of_MNR }

FRA_election_state_check = {
	if = {
		limit = {
			check_variable = { FRA_election_states_var > 4 }
		}
		set_variable = { FRA_election_states_right_var = FRA_election_states_var }
		add_to_variable = { FRA_election_states_right_var = -4 }
	}
	ROOT = { FRA_ideology_calculation = yes }
}
