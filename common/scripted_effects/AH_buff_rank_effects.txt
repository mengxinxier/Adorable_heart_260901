AH_br_rebuild_and_sort = {
	clear_array = global.countries_buff
	set_variable = { global.countries_buff_index = 0 }
	every_country = {
		limit = {
			OR = {
				NOT = { check_variable = { global.AH_br_filter_major = 1 } }
				is_major = yes
				is_ai = no
				num_of_factories > 29
			}
		}
		AH_br_assign_sort_score = yes
		add_to_array = { global.countries_buff = THIS }
	}
	get_sorted_scored_countries = {
		scorer = Buff_rank_scorer
		array = global.countries_buff
	}
	for_each_scope_loop = {
		array = global.countries_buff
		add_to_variable = { global.countries_buff_index = 1 }
		set_variable = { econ_ranking = global.countries_buff_index }
	}
	clear_variable = global.countries_buff_index
	add_to_variable = { global.AH_br_dirty = 1 }
}

#点击列头切换排序（同列翻转升降序，新列默认降序）
AH_br_click_sort_by_id = {
	if = {
		limit = { check_variable = { global.AH_br_sort_id = _target_id } }
		if = { limit = { check_variable = { global.AH_br_sort_asc = 1 } } set_variable = { global.AH_br_sort_asc = 0 } }
		else = { set_variable = { global.AH_br_sort_asc = 1 } }
	}
	else = {
		set_variable = { global.AH_br_sort_id = _target_id }
		set_variable = { global.AH_br_sort_asc = 0 }
	}
	AH_br_rebuild_and_sort = yes
}

	set_variable = { global.AH_br_inspect_divisions = num_divisions }

	clear_array = global.br_v
	clear_array = global.br_k
	clear_array = global.br_d
	AH_br_read_inspect_total = yes
	add_to_variable = { global.AH_br_dirty = 1 }
}

#从政治界面的调试按钮打开或关闭总览窗口（首次打开自动聚焦玩家国第1词条）
AH_br_toggle_window_open = {
	if = {
		limit = { has_country_flag = Adorable_Heart_buff_rank_window_visible }
		clr_country_flag = Adorable_Heart_buff_rank_window_visible
	}
	else = {
		set_country_flag = Adorable_Heart_buff_rank_window_visible
		if = { limit = { NOT = { has_variable = global.TheresaBot_modifier_var } } set_variable = { global.TheresaBot_modifier_var = 0 } }
		if = { limit = { NOT = { has_variable = global.AH_br_subpage_var } } set_variable = { global.AH_br_subpage_var = 0 } }
		if = { limit = { NOT = { has_variable = global.AH_br_sort_id } } set_variable = { global.AH_br_sort_id = 0 } }
		if = { limit = { NOT = { has_variable = global.AH_br_sort_asc } } set_variable = { global.AH_br_sort_asc = 0 } }
		if = { limit = { NOT = { has_variable = global.AH_br_filter_major } } set_variable = { global.AH_br_filter_major = 0 } }
		if = { limit = { check_variable = { global.AH_br_inspect_mod_id < 1 } } set_variable = { global.AH_br_inspect_mod_id = 1 } }
		AH_br_sync_ui_frames = yes
		AH_br_rebuild_and_sort = yes
		AH_br_inspect_this_country = yes
	}
}
