GER_Compute_European_Union = {
	set_variable = { global.num_of_GER_European_Union_Vote_support = 0 }
	set_variable = { global.num_of_GER_European_Union_Vote_oppose = 0 }
	set_variable = { global.num_of_GER_European_Union_Vote_neutrality = 0 }
	set_variable = { global.num_of_GER_European_Union_Vote_nothing = 0 }
	set_variable = { global.num_of_GER_European_Union_all_vote = 0 }
	set_variable = { global.num_of_GER_European_Union_half_vote = 0 }
	for_each_scope_loop = {
		array = global.GER_European_Union_members_array
		if = {
			limit = { check_variable = { GER_European_Union_attitude_VAR = 4 } }
			add_to_variable = { global.num_of_GER_European_Union_Vote_support = 1 }
			set_variable = { GER_European_Union_Vote_VAR = 4 }
			if = {

build_European_Union_effect = {
	custom_effect_tooltip = FRA_Start_integration_process_tt  ##吞并所有成员国并获得核心\n获得所有欧洲领土的宣称
	if = {  ##科研槽不足7则+1
		limit = { amount_research_slots < 7 }
		add_research_slot = 1
	}
	set_cosmetic_tag = EUR_UNIFIED  ##cosmetic tag
	hidden_effect = {
		every_country = {  ##吞并所有成员国并获得核心
			limit = {
				is_subject_of = ROOT
				capital_scope = { is_on_continent = europe }
			}
			if = {
				limit = {
					ROOT = { is_ai = yes }
					is_ai = no
				}
				ROOT = { change_tag_from_PREV = yes }
			}
			every_core_state = {
				add_core_of = ROOT
			}
			ROOT = { annex_country = { target = PREV transfer_troops = yes } }
		}
		for_each_scope_loop = {
			array = global.GER_European_Union_members_array
			every_subject_country = {  ##附庸
				if = {
					limit = { capital_scope = { is_on_continent = europe } }  ##欧洲国家
					if = {
						limit = {
							ROOT = { is_ai = yes }
							is_ai = no
						}
						ROOT = { change_tag_from_PREV = yes }
					}
					every_core_state = {
						add_core_of = ROOT
					}
					every_character = {
						limit = { is_general_advisor = no }
						set_nationality = ROOT
					}
					ROOT = { annex_country = { target = PREV transfer_troops = yes } }
				}
				else = {
					ROOT = { puppet = PREV }  ##非欧洲则附庸
				}
			}
			if = {
				limit = {
					not = { tag = ROOT }
					capital_scope = { is_on_continent = europe }
				}
				if = {
					limit = {
						ROOT = { is_ai = yes }
						is_ai = no
					}
					ROOT = { change_tag_from_PREV = yes }
				}
				every_core_state = {
					add_core_of = ROOT
				}
				every_character = {
					limit = { is_general_advisor = no }
					set_nationality = ROOT
				}
				ROOT = { annex_country = { target = PREV transfer_troops = yes } }
			}
		}
		every_state = {  ##其他欧洲领土添加宣称与造核
			limit = {
				is_on_continent = europe
				NOT = {
					or = {
						is_core_of = ROOT
						is_claimed_by = ROOT
					}
				}
			}
			add_claim_by = ROOT
			enable_state_integration_50_compliance = yes
		}
		news_event = {  ##欧盟成立的新闻
			id = FRA_Gensokyo_news.013
		}
		country_event = {  ##新欧洲的首都
			id = ah_germany.106
		}
		clear_array = global.GER_European_Union_members_array
	}
}
