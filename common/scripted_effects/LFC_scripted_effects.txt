	#每月调度许可获取量计算
	LFC_Preset_monthly_license_calculation = {
		# 读取当前可用工厂数量（不含已贡献的部分）
		set_variable = {
			var = LFC_VAR_temp_civ_factories
			value = num_of_civilian_factories
		}
		set_variable = {
			var = LFC_VAR_temp_mil_factories
			value = num_of_military_factories
		}

		# 基础许可 + 贡献工厂计算（根据经济联系等级）
		if = {
			limit = {
				has_idea = LFC_idea_jjlx_1
			}
			set_variable = {
				LFC_VAR_monthly_license_base = 30
			}
			# 共享10%，贡献民工 = 可用民工 × 0.1 / 0.9（反推贡献前总量）
			set_variable = {
				var = LFC_VAR_jjlx_contributed_civ
				value = LFC_VAR_temp_civ_factories
			}
			multiply_variable = {
				var = LFC_VAR_jjlx_contributed_civ
				value = 0.1
			}
			divide_variable = {
				var = LFC_VAR_jjlx_contributed_civ
				value = 0.9
			}
			round_variable = LFC_VAR_jjlx_contributed_civ
			# 贡献军工 = 可用军工 × 0.1 / 0.9（反推贡献前总量）
			set_variable = {
				var = LFC_VAR_jjlx_contributed_mil
				value = LFC_VAR_temp_mil_factories
			}
			multiply_variable = {
				var = LFC_VAR_jjlx_contributed_mil
				value = 0.1
			}
			divide_variable = {
				var = LFC_VAR_jjlx_contributed_mil
				value = 0.9
			}
			round_variable = LFC_VAR_jjlx_contributed_mil
		}
		else_if = {
			limit = {
                has_idea = LFC_idea_jjlx_2		
			}
			set_variable = {
				LFC_VAR_monthly_license_base = 60
			}
			# 共享20%，贡献民工 = 可用民工 × 0.2 / 0.8（反推贡献前总量）
			set_variable = {
				var = LFC_VAR_jjlx_contributed_civ
				value = LFC_VAR_temp_civ_factories
			}
			multiply_variable = {
				var = LFC_VAR_jjlx_contributed_civ
				value = 0.2
			}
			divide_variable = {
				var = LFC_VAR_jjlx_contributed_civ
				value = 0.8
			}
			round_variable = LFC_VAR_jjlx_contributed_civ
			# 贡献军工 = 可用军工 × 0.2 / 0.8（反推贡献前总量）
			set_variable = {
				var = LFC_VAR_jjlx_contributed_mil
				value = LFC_VAR_temp_mil_factories
			}

				LFC_VAR_monthly_license_base = 90
			}

			set_variable = {
				var = LFC_VAR_jjlx_contributed_civ
				value = LFC_VAR_temp_civ_factories
			}
			multiply_variable = {
				var = LFC_VAR_jjlx_contributed_civ
				value = 0.3
			}
			divide_variable = {
				var = LFC_VAR_jjlx_contributed_civ
				value = 0.7
			}
			round_variable = LFC_VAR_jjlx_contributed_civ
			# 贡献军工 = 可用军工 × 0.3 / 0.7（反推贡献前总量）
			set_variable = {
				var = LFC_VAR_jjlx_contributed_mil
				value = LFC_VAR_temp_mil_factories
			}
			multiply_variable = {
				var = LFC_VAR_jjlx_contributed_mil
				value = 0.3
			}

				var = LFC_VAR_jjlx_contributed_mil

		}

			set_variable = {
				LFC_VAR_jjlx_contributed_mil = 0
			}
		}

			var = LFC_VAR_jjlx_factory_count
			value = LFC_VAR_jjlx_contributed_civ
		}
		add_to_variable = {

			value = LFC_VAR_jjlx_contributed_mil
		}

		}

		}

			value = LFC_VAR_monthly_license_base
		}
		add_to_variable = {

			value = LFC_VAR_monthly_license_factory_bonus
		}

			LFC_frontier_cost_manpower = 1000

		set_variable = {
			LFC_frontier_cost_rifles = LFC_frontier_expedition_scale
		}
		multiply_variable = {
			LFC_frontier_cost_rifles = 500
		}
		set_variable = {
			LFC_frontier_cost_support = LFC_frontier_expedition_scale
		}
		multiply_variable = {
			LFC_frontier_cost_support = 20
		}
		set_variable = {
			LFC_frontier_cost_trucks = 0
		}

			LFC_frontier_cost_knights = 0

		LFC_frontier_initialize_expedition_draft = yes

				LFC_frontier_expedition_can_depart = yes
			}
			set_variable = {
				VAR_LFC_Rainforest_state = LFC_frontier_selected_state

			}

				LFC_VAR_Rainforest_supply = LFC_frontier_draft_supply
			}
			set_variable = {
				LFC_VAR_Rainforest_manpower = LFC_frontier_cost_manpower
			}
			set_variable = {
				LFC_VAR_Rainforest_progress = 0
			}
			set_variable = {
				LFC_frontier_departure_supply = LFC_frontier_draft_supply
			}
			set_variable = {
				LFC_frontier_departure_manpower = LFC_frontier_cost_manpower
			}
			set_temp_variable = {
				LFC_frontier_expedition_payment = LFC_frontier_draft_supply
			}
			multiply_temp_variable = {
				LFC_frontier_expedition_payment = -1
			}

			}

			}
			multiply_temp_variable = {
				LFC_frontier_expedition_payment = -1
			}
			add_manpower = LFC_frontier_expedition_payment
			set_temp_variable = {
				LFC_frontier_expedition_payment = LFC_frontier_cost_rifles
			}
			multiply_temp_variable = {
				LFC_frontier_expedition_payment = -1
			}
			add_equipment_to_stockpile = {
				type = infantry_equipment
				amount = LFC_frontier_expedition_payment
			}

				LFC_frontier_expedition_payment = LFC_frontier_cost_support
			}
			multiply_temp_variable = {
				LFC_frontier_expedition_payment = -1
			}
			add_equipment_to_stockpile = {
				type = support_equipment
				amount = LFC_frontier_expedition_payment
			}
			if = {
				limit = {
					check_variable = { LFC_frontier_cost_trucks > 0 }

					LFC_frontier_expedition_payment = -1

					type = motorized_equipment
					amount = LFC_frontier_expedition_payment
				}
			}

				limit = {

				}

					LFC_frontier_expedition_payment = LFC_frontier_cost_knights

				multiply_temp_variable = {
					LFC_frontier_expedition_payment = -1
				}
				add_equipment_to_stockpile = {
					type = construct_knight_equipment
					amount = LFC_frontier_expedition_payment
				}

			set_country_flag = FLAG_LFC_Rainforest_Acting
