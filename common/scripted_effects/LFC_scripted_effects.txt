	#每月调度许可获取量计算
	LFC_Preset_monthly_license_calculation = {
		# 读取当前可用工厂数量（不含已贡献的部分）
		set_variable = {
			var = LFC_VAR_temp_civ_factories
			value = num_of_civilian_factories
		}
		set_variable = {
			var = LFC_VAR_temp_mil_factories
			value = num_of_military_factories
		}

		# 基础许可 + 贡献工厂计算（根据经济联系等级）
		if = {
			limit = {
				has_idea = LFC_idea_jjlx_1
			}
			set_variable = {
				LFC_VAR_monthly_license_base = 30
			}
			# 共享10%，贡献民工 = 可用民工 × 0.1 / 0.9（反推贡献前总量）
			set_variable = {
				var = LFC_VAR_jjlx_contributed_civ
				value = LFC_VAR_temp_civ_factories
			}
			multiply_variable = {
				var = LFC_VAR_jjlx_contributed_civ
				value = 0.1
			}
			divide_variable = {
				var = LFC_VAR_jjlx_contributed_civ
				value = 0.9
			}
			round_variable = LFC_VAR_jjlx_contributed_civ
			# 贡献军工 = 可用军工 × 0.1 / 0.9（反推贡献前总量）
			set_variable = {
				var = LFC_VAR_jjlx_contributed_mil
				value = LFC_VAR_temp_mil_factories
			}
			multiply_variable = {
				var = LFC_VAR_jjlx_contributed_mil
				value = 0.1
			}
			divide_variable = {
				var = LFC_VAR_jjlx_contributed_mil
				value = 0.9
			}
			round_variable = LFC_VAR_jjlx_contributed_mil
		}
		else_if = {
			limit = {
                has_idea = LFC_idea_jjlx_2		
			}
			set_variable = {
				LFC_VAR_monthly_license_base = 60
			}
			# 共享20%，贡献民工 = 可用民工 × 0.2 / 0.8（反推贡献前总量）
			set_variable = {
				var = LFC_VAR_jjlx_contributed_civ
				value = LFC_VAR_temp_civ_factories
			}
			multiply_variable = {
				var = LFC_VAR_jjlx_contributed_civ
				value = 0.2
			}
			divide_variable = {
				var = LFC_VAR_jjlx_contributed_civ
				value = 0.8
			}
			round_variable = LFC_VAR_jjlx_contributed_civ
			# 贡献军工 = 可用军工 × 0.2 / 0.8（反推贡献前总量）
			set_variable = {
				var = LFC_VAR_jjlx_contributed_mil
				value = LFC_VAR_temp_mil_factories
			}
			multiply_variable = {
				var = LFC_VAR_jjlx_contributed_mil
				value = 0.2
			}
			divide_variable = {
				var = LFC_VAR_jjlx_contributed_mil
				value = 0.8
			}
			round_variable = LFC_VAR_jjlx_contributed_mil
		}
		else_if = {
			limit = {
				has_idea = LFC_idea_jjlx_3
			}
			set_variable = {
				LFC_VAR_monthly_license_base = 90
			}
			# 共享30%，贡献民工 = 可用民工 × 0.3 / 0.7（反推贡献前总量）
			set_variable = {
				var = LFC_VAR_jjlx_contributed_civ
				value = LFC_VAR_temp_civ_factories
			}
			multiply_variable = {
				var = LFC_VAR_jjlx_contributed_civ
				value = 0.3
			}
			divide_variable = {
				var = LFC_VAR_jjlx_contributed_civ
				value = 0.7
			}
			round_variable = LFC_VAR_jjlx_contributed_civ
			# 贡献军工 = 可用军工 × 0.3 / 0.7（反推贡献前总量）
			set_variable = {
				var = LFC_VAR_jjlx_contributed_mil
				value = LFC_VAR_temp_mil_factories
			}
			multiply_variable = {
				var = LFC_VAR_jjlx_contributed_mil
				value = 0.3
			}
			divide_variable = {
				var = LFC_VAR_jjlx_contributed_mil
				value = 0.7
			}
			round_variable = LFC_VAR_jjlx_contributed_mil
		}
		else = {
			set_variable = {
				LFC_VAR_monthly_license_base = 0
			}
			set_variable = {
				LFC_VAR_jjlx_contributed_civ = 0
			}
			set_variable = {
				LFC_VAR_jjlx_contributed_mil = 0
			}
		}

		# 合计贡献工厂数 = 民工 + 军工
		set_variable = {
			var = LFC_VAR_jjlx_factory_count
			value = LFC_VAR_jjlx_contributed_civ
		}
		add_to_variable = {
			var = LFC_VAR_jjlx_factory_count
			value = LFC_VAR_jjlx_contributed_mil
		}

		# 工厂贡献加成：每1工厂额外+2许可
		set_variable = {
			var = LFC_VAR_monthly_license_factory_bonus
			value = LFC_VAR_jjlx_factory_count
		}
		multiply_variable = {
			var = LFC_VAR_monthly_license_factory_bonus
			value = 2
		}

		# 合计月度获取量 = 基础 + 工厂加成
		set_variable = {
			var = LFC_VAR_monthly_license_total
			value = LFC_VAR_monthly_license_base
		}
		add_to_variable = {
			var = LFC_VAR_monthly_license_total
			value = LFC_VAR_monthly_license_factory_bonus
		}

		# 计算月度许可增长
        set_variable = {  #归零累加器
            var = LFC_VAR_Empire_license_monthly_end
            value = 0
        }
        add_to_variable = {  #计入基础计算值（基础许可+工厂贡献加成）
            LFC_VAR_Empire_license_monthly_end = LFC_VAR_monthly_license_total
        }
        add_to_variable = {  #计入自定义固定值修正
            LFC_VAR_Empire_license_monthly_end = modifier@LFC_license_monthly_add
        }

        set_variable = {  #读取自定义百分比修正总和
            var = LFC_VAR_Empire_license_monthly_factor_0
            value = modifier@LFC_license_monthly_factor
        }
        add_to_variable = {  #转为乘法系数：1 + 修正总和（如修正0.15 → 系数1.15）
            LFC_VAR_Empire_license_monthly_factor_0 = 1
        }
        multiply_variable = {  #百分比修正套用到最终月增长
			var = LFC_VAR_Empire_license_monthly_end
			value = LFC_VAR_Empire_license_monthly_factor_0
		}


	}
