PAU_refresh_mil_reform_tiers = {
    if = {
        limit = { NOT = { has_country_flag = PAU_reform_reset_initialized } }
        set_country_flag = PAU_reform_reset_initialized
        set_variable = { PAU_reform_reset_count = 1 }
    }
    clamp_variable = { var = PAU_mil_reform_progress min = 0 max = 36 }
    if = {
        limit = {
            check_variable = { PAU_mil_reform_progress > 35 }
            NOT = { has_country_flag = PAU_mil_reform_cap_reward_claimed }
        }
        set_country_flag = PAU_mil_reform_cap_reward_claimed
        add_to_variable = { PAU_reform_reset_count = 1 }
        country_event = { id = PAU.701 hours = 1 }
    }
    set_variable = { PAU_mil_reform_progress_frame = 0 }
    if = {
        limit = { check_variable = { PAU_mil_reform_progress > 0 } }
        set_variable = { PAU_mil_reform_progress_frame = PAU_mil_reform_progress }
        divide_variable = { PAU_mil_reform_progress_frame = 36 }
        multiply_variable = { PAU_mil_reform_progress_frame = 100 }
    }
    if = {
        limit = {
            NOT = { has_dynamic_modifier = { modifier = PAU_mil_reform } }
            OR = {
                check_variable = { PAU_mil_reform_progress > 0 }
                check_variable = { PAU_axia_mil > 0 }
                check_variable = { PAU_xfxy_mil > 0 }
                check_variable = { PAU_xiaoyao_mil > 0 }
            }
        }
        add_dynamic_modifier = { modifier = PAU_mil_reform }

PAU_refresh_harbor_spillover_states = {
    if = {
        limit = { NOT = { has_variable = PAU_harbor_dev } }
        set_variable = { PAU_harbor_dev = 0 }  #避风港发展度=每次扩建+1，上限6
    }
    every_controlled_state = {
        limit = { has_state_flag = PAU_safe_harbor }  #州已指定为避风港标记
        if = {
            limit = { has_dynamic_modifier = { modifier = PAU_harbor_spillover_1 } }
            remove_dynamic_modifier = { modifier = PAU_harbor_spillover_1 }
        }
        if = {
            limit = { has_dynamic_modifier = { modifier = PAU_harbor_spillover_2 } }
            remove_dynamic_modifier = { modifier = PAU_harbor_spillover_2 }
        }
        if = {
            limit = { has_dynamic_modifier = { modifier = PAU_harbor_spillover_3 } }
            remove_dynamic_modifier = { modifier = PAU_harbor_spillover_3 }
        }
        if = {
            limit = { has_dynamic_modifier = { modifier = PAU_harbor_spillover_4 } }
            remove_dynamic_modifier = { modifier = PAU_harbor_spillover_4 }
        }
        if = {
            limit = { check_variable = { PAU_harbor_dev > 5 } }
            add_dynamic_modifier = { modifier = PAU_harbor_spillover_4 }
        }
        else_if = {
            limit = { check_variable = { PAU_harbor_dev > 3 } }

    }

PAU_pull_into_white_night_faction = {
    PAU = {
        if = {
            limit = { NOT = { is_in_faction = yes } }
            create_faction_from_template = faction_template_nmgr
            set_faction_leader = yes
        }
    }
    if = {
        limit = {
            PAU = { is_in_faction = yes }
            ROOT = { NOT = { is_in_faction_with = PAU } }
        }
        if = {
            limit = { ROOT = { is_in_faction = yes } }
            if = {
                limit = {
                    ROOT = {
                        NOT = { has_country_flag = PAU_allow_major_faction_pull }
                        OR = {
                            faction_leader = { OR = { tag = GER tag = DVL tag = ENG tag = SOV } }
                            has_faction_template = faction_template_axis
                            has_faction_template = faction_template_allies
                            has_faction_template = faction_template_comintern
                        }
                    }
                }
                ROOT = { leave_faction = yes }
                PAU = { faction_leader = { add_to_faction = ROOT } }
            }
            else = {
                ROOT = {
                    every_faction_member = {
                        display_individual_scopes = yes
                        leave_faction = yes
                        PAU = {
                            faction_leader = {
                                add_to_faction = PREV.PREV
                            }
                        }
                    }
                }
            }
        }
        else = {
            PAU = { faction_leader = { add_to_faction = ROOT } }
        }
    }
    ROOT = { clr_country_flag = PAU_allow_major_faction_pull }
}

# ==================== 白夜决议栏 GUI 变量同步与每日计时 ====================
PAU_sync_baiye_gui_vars = {
    if = {
        limit = { NOT = { has_variable = PAU_baiye_patrol_count } }
        set_variable = { PAU_baiye_patrol_count = 0 }
    }
    if = {
        limit = { NOT = { has_variable = PAU_baiye_story_stage } }
        set_variable = { PAU_baiye_story_stage = 0 }
    }
    set_variable = { PAU_baiye_story_pct = PAU_baiye_story_stage }
    multiply_variable = { PAU_baiye_story_pct = 20 }

    if = {
        limit = {
            has_country_flag = PAU_baiye_morning_run_done
            has_character = PAU_baiye
            PAU_baiye = { NOT = { has_trait = PAU_baiye_fit } }
        }
        clr_country_flag = PAU_baiye_morning_run_done
    }
    if = {
        limit = {
            NOT = { has_country_flag = PAU_baiye_morning_run_done }
            has_character = PAU_baiye
            PAU_baiye = { has_trait = PAU_baiye_fit }
        }
        set_country_flag = PAU_baiye_morning_run_done
        clr_country_flag = PAU_baiye_morning_run_active
    }
    if = {
        limit = {
            NOT = { has_country_flag = PAU_baiye_shrine_done }
            OR = {
                has_idea = PAU_baiye_shrine_fish_treat_idea
                has_idea = PAU_baiye_shrine_order_idea

        set_country_flag = PAU_baiye_shrine_done

    if = {
        limit = {
            NOT = { has_country_flag = PAU_baiye_sakura_done }

        }
        set_country_flag = PAU_baiye_sakura_done

    }
}

PAU_daily_baiye_tick = {
    if = {
        limit = {
            has_country_flag = PAU_baiye_morning_run_active
            has_variable = PAU_baiye_morning_run_timer
        }
        if = {
            limit = { check_variable = { PAU_baiye_morning_run_timer > 0 } }
            subtract_from_variable = { PAU_baiye_morning_run_timer = 1 }
        }
        if = {
            limit = { check_variable = { PAU_baiye_morning_run_timer < 1 } }
            clear_variable = PAU_baiye_morning_run_timer
            clr_country_flag = PAU_baiye_morning_run_active
            set_country_flag = PAU_baiye_morning_run_done
            if = {
                limit = {
                    has_character = PAU_baiye
                    PAU_baiye = { has_trait = PAU_baiye_chubby }
                }
                PAU_baiye = {
                    remove_unit_leader_trait = PAU_baiye_chubby
                    add_unit_leader_trait = PAU_baiye_fit
                    add_attack = 2
                    add_defense = 2
                    add_planning = 2
                    add_logistics = 2
                }
            }
            country_event = { id = PAU.7013 }
        }
    }

    if = {
        limit = { check_variable = { PAU_baiye_mist_active > 0 } }
        subtract_from_variable = { PAU_baiye_mist_active = 1 }
    }
    else_if = {
        limit = { check_variable = { PAU_baiye_mist_cooldown > 0 } }
        subtract_from_variable = { PAU_baiye_mist_cooldown = 1 }
    }

    if = {
        limit = { check_variable = { PAU_baiye_lookout_active > 0 } }
        subtract_from_variable = { PAU_baiye_lookout_active = 1 }
    }
    else_if = {
        limit = { check_variable = { PAU_baiye_lookout_cooldown > 0 } }
        subtract_from_variable = { PAU_baiye_lookout_cooldown = 1 }
    }

    if = {
        limit = {
            has_country_flag = PAU_baiye_sakura_active
            has_variable = PAU_baiye_sakura_timer
        }
        if = {
            limit = { check_variable = { PAU_baiye_sakura_timer > 0 } }
            subtract_from_variable = { PAU_baiye_sakura_timer = 1 }
        }
        if = {
            limit = { check_variable = { PAU_baiye_sakura_timer < 1 } }
            clear_variable = PAU_baiye_sakura_timer
            clr_country_flag = PAU_baiye_sakura_active
            set_country_flag = PAU_baiye_sakura_done
            country_event = { id = PAU.7012 }
        }
    }

    PAU_sync_baiye_gui_vars = yes
}
