PAU_refresh_harbor_spillover_states = {
    if = {
        limit = { NOT = { has_variable = PAU_harbor_dev } }
        set_variable = { PAU_harbor_dev = 0 }  #避风港发展度=每次扩建+1，上限6
    }
    every_controlled_state = {
        limit = { has_state_flag = PAU_safe_harbor }  #州已指定为避风港标记
        if = {
            limit = { has_dynamic_modifier = { modifier = PAU_harbor_spillover_1 } }
            remove_dynamic_modifier = { modifier = PAU_harbor_spillover_1 }
        }
        if = {
            limit = { has_dynamic_modifier = { modifier = PAU_harbor_spillover_2 } }
            remove_dynamic_modifier = { modifier = PAU_harbor_spillover_2 }
        }
        if = {
            limit = { has_dynamic_modifier = { modifier = PAU_harbor_spillover_3 } }
            remove_dynamic_modifier = { modifier = PAU_harbor_spillover_3 }
        }
        if = {
            limit = { has_dynamic_modifier = { modifier = PAU_harbor_spillover_4 } }
            remove_dynamic_modifier = { modifier = PAU_harbor_spillover_4 }
        }
        if = {
            limit = { check_variable = { PAU_harbor_dev > 5 } }
            add_dynamic_modifier = { modifier = PAU_harbor_spillover_4 }
        }
        else_if = {
            limit = { check_variable = { PAU_harbor_dev > 3 } }
            add_dynamic_modifier = { modifier = PAU_harbor_spillover_3 }
        }
        else_if = {
            limit = { check_variable = { PAU_harbor_dev > 1 } }
            add_dynamic_modifier = { modifier = PAU_harbor_spillover_2 }
        }
        else_if = {
            limit = { check_variable = { PAU_harbor_dev > 0 } }
            add_dynamic_modifier = { modifier = PAU_harbor_spillover_1 }
        }
    }
}

PAU_pull_into_white_night_faction = {
    PAU = {
        if = {
            limit = { NOT = { is_in_faction = yes } }
            create_faction_from_template = faction_template_nmgr
            set_faction_leader = yes
        }
    }
    if = {
        limit = {
            PAU = { is_in_faction = yes }
            ROOT = { NOT = { is_in_faction_with = PAU } }
        }
        if = {
            limit = { ROOT = { is_in_faction = yes } }
            if = {
                limit = {
                    ROOT = {
                        NOT = { has_country_flag = PAU_allow_major_faction_pull }
                        OR = {
                            faction_leader = { OR = { tag = GER tag = DVL tag = ENG tag = SOV } }
                            has_faction_template = faction_template_axis
                            has_faction_template = faction_template_allies
                            has_faction_template = faction_template_comintern
                        }
                    }
                }
                ROOT = { leave_faction = yes }
                PAU = { faction_leader = { add_to_faction = ROOT } }
            }
            else = {
                ROOT = {
                    every_faction_member = {
                        display_individual_scopes = yes
                        leave_faction = yes
                        PAU = {
                            faction_leader = {
                                add_to_faction = PREV.PREV
                            }
                        }
                    }
                }
            }
        }
        else = {
            PAU = { faction_leader = { add_to_faction = ROOT } }
        }
    }
    ROOT = { clr_country_flag = PAU_allow_major_faction_pull }
}

PAU_advance_baiye_story = {
    if = { limit = { NOT = { has_variable = PAU_baiye_patrol_count } } set_variable = { PAU_baiye_patrol_count = 0 } }  #白夜巡游计数=每巡游一个州+1
    if = { limit = { NOT = { has_variable = PAU_baiye_story_stage } } set_variable = { PAU_baiye_story_stage = 0 } }  #白夜巡游剧情阶段=巡游计数到1/2/3/4时依次+1
    if = {
        limit = {
            has_country_flag = PAU_undead_defeated  #亡灵战争结束标记
            NOT = { check_variable = { PAU_baiye_story_stage > 4 } }
        }
        set_variable = { PAU_baiye_story_stage = 5 }
    }
    add_to_variable = { PAU_baiye_patrol_count = 1 }
    if = {
        limit = {
            check_variable = { PAU_baiye_patrol_count > 0 }
            NOT = { check_variable = { PAU_baiye_story_stage > 0 } }
        }
        set_variable = { PAU_baiye_story_stage = 1 }
        country_event = { id = PAU.7001 hours = 6 }
    }
    if = {
        limit = {
            check_variable = { PAU_baiye_patrol_count > 1 }
            NOT = { check_variable = { PAU_baiye_story_stage > 1 } }
        }
        set_variable = { PAU_baiye_story_stage = 2 }
        country_event = { id = PAU.7002 hours = 6 }
    }
    if = {
        limit = {
            check_variable = { PAU_baiye_patrol_count > 2 }
            NOT = { check_variable = { PAU_baiye_story_stage > 2 } }
        }
        set_variable = { PAU_baiye_story_stage = 3 }
        country_event = { id = PAU.7003 hours = 6 }
    }
    if = {
        limit = {
            check_variable = { PAU_baiye_patrol_count > 3 }
            NOT = { check_variable = { PAU_baiye_story_stage > 3 } }
        }
        set_variable = { PAU_baiye_story_stage = 4 }
        country_event = { id = PAU.7004 hours = 6 }
    }
}
