PAU_sync_desk_plum_frame = {
    set_temp_variable = { PAU_plum_total = 0 }
    if = {
        limit = { has_variable = PAU_mil_reform_progress }
        add_to_temp_variable = { PAU_plum_total = PAU_mil_reform_progress }
    }
    if = {
        limit = { has_variable = PAU_eco_reform_progress }
        add_to_temp_variable = { PAU_plum_total = PAU_eco_reform_progress }
    }
    set_variable = { PAU_desk_plum_frame = 1 }
    if = {
        limit = { check_variable = { PAU_plum_total > 49 } }
        set_variable = { PAU_desk_plum_frame = 5 }
    }
    else_if = {
        limit = { check_variable = { PAU_plum_total > 29 } }
        set_variable = { PAU_desk_plum_frame = 4 }
    }
    else_if = {
        limit = { check_variable = { PAU_plum_total > 13 } }
        set_variable = { PAU_desk_plum_frame = 3 }
    }
    else_if = {
        limit = { check_variable = { PAU_plum_total > 0 } }
        set_variable = { PAU_desk_plum_frame = 2 }
    }
}

PAU_pull_into_white_night_faction = {
    PAU = {
        if = {
            limit = { NOT = { is_in_faction = yes } }
            create_faction_from_template = faction_template_nmgr
            set_faction_leader = yes
        }
    }
    if = {
        limit = {
            PAU = { is_in_faction = yes }
            ROOT = { NOT = { is_in_faction_with = PAU } }
        }
        if = {
            limit = { ROOT = { is_in_faction = yes } }
            if = {
                limit = {
                    ROOT = {
                        OR = {
                            has_country_flag = PAU_pull_single_country_only
                            tag = MGU
                            tag = LFC
                            AND = {
                                NOT = { has_country_flag = PAU_allow_major_faction_pull }
                                OR = {
                                    faction_leader = { OR = { tag = GER tag = DVL tag = ENG tag = SOV } }
                                    has_faction_template = faction_template_axis
                                    has_faction_template = faction_template_allies
                                    has_faction_template = faction_template_comintern
                                }
                            }
                        }
                    }
                }
                hidden_effect = {
                    ROOT = { set_country_flag = leave_faction_happily_flag }
                }
                ROOT = { leave_faction = yes }
                PAU = { faction_leader = { add_to_faction = ROOT } }
                hidden_effect = {
                    ROOT = { clr_country_flag = leave_faction_happily_flag }
                }
            }
            else = {
                ROOT = {
                    every_faction_member = {
                        display_individual_scopes = yes
                        leave_faction = yes
                        PAU = {
                            faction_leader = {
                                add_to_faction = PREV.PREV
                            }
                        }
                    }
                }
            }
        }
        else = {
            PAU = { faction_leader = { add_to_faction = ROOT } }
        }
    }
    ROOT = {
        clr_country_flag = PAU_allow_major_faction_pull
        clr_country_flag = PAU_pull_single_country_only
    }
}

PAU_sync_south_america_all = {
    PAU = {
        clamp_variable = { var = PAU_south_america_development min = 0 max = 100 }
        if = {
            limit = { NOT = {
                tag = PAU
                has_dynamic_modifier = { modifier = PAU_south_america_unity }
            }
        }
        add_dynamic_modifier = { modifier = PAU_south_america_unity }
    }
    if = {
        limit = { is_in_faction = yes }
        every_faction_member = {
            limit = {
                NOT = { tag = PAU }
                capital_scope = { is_on_continent = south_america }
            }
            if = {

#切换阵营保持联合发展计划
PAU_recover_dev_program_loop = {
    PAU = {
        if = {
            limit = {
                has_country_flag = PAU_south_america_super_fired
                has_country_flag = PAU_dev_program_auto_loop
                is_subject = no
            }
            if = {
                limit = { NOT = { has_active_mission = PAU_south_america_development_program_loop } }
                activate_mission = PAU_south_america_development_program_loop
            }
            if = {
                limit = { is_in_faction = yes }
                if = {
                    limit = { has_variable = PAU_dev_program_current_leader }
                    var:PAU.PAU_dev_program_current_leader = {
                        if = {
                            limit = {
                                exists = yes
                                OR = {
                                    tag = PAU
                                    is_in_faction_with = PAU
                                }
                            }
                            PAU_distribute_dev_program_idea_from_scope = yes
                        }
                        else = {
                            PAU = {
                                set_variable = { PAU_dev_program_current_leader = PAU }
                                PAU_distribute_dev_program_idea_from_scope = yes
                            }
                        }
                    }
                }
                else_if = {
                    limit = {
                        NOT = {
                            any_other_country = {
                                is_in_faction_with = PAU
                                capital_scope = { is_on_continent = south_america }
                                OR = {
                                    has_idea = PAU_south_america_development_program_ideaPAU
                                    has_idea = PAU_south_america_development_program_ideaPA1
                                    has_idea = PAU_south_america_development_program_ideaPAQ
                                    has_idea = PAU_south_america_development_program_ideaELF
                                    has_idea = PAU_south_america_development_program_ideaLFC
                                    has_idea = PAU_south_america_development_program_ideaMGU
                                    has_idea = PAU_south_america_development_program_ideaPGM
                                    has_idea = PAU_south_america_development_program_ideaAHM
                                }
                            }
                        }
                    }
                    set_country_flag = pau_development_program_leader
                    set_variable = { PAU_dev_program_current_leader = PAU }
                    PAU_distribute_dev_program_idea_from_scope = yes
                }
            }
        }
    }
}
