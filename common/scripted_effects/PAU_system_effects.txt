#PAU_pull_into_white_night_faction：把目标国拉入白夜同盟阵营
PAU_pull_into_white_night_faction = {
    #确保白夜同盟存在；普通阵营可整块带入，轴心/魔导/圣杯/苏联主导阵营只接收直接目标国
    PAU = {
        if = {
            limit = { NOT = { is_in_faction = yes } }
            create_faction_from_template = faction_template_nmgr
            set_faction_leader = yes
            set_faction_name = PAU_white_night_alliance
            set_faction_manifest = PAU_faction_manifest_white_night
        }
    }
    if = {
        limit = {
            PAU = { is_in_faction = yes }
            ROOT = { NOT = { is_in_faction_with = PAU } }
        }
        if = {
            limit = { ROOT = { is_in_faction = yes } }
            if = {
                limit = {
                    ROOT = {
                        NOT = { has_country_flag = PAU_allow_major_faction_pull }
                        OR = {
                            faction_leader = { OR = { tag = GER tag = DVL tag = ENG tag = SOV } }
                            has_faction_template = faction_template_axis
                            has_faction_template = faction_template_allies
                            has_faction_template = faction_template_comintern
                        }
                    }
                }
                #轴心国/魔导国/圣杯/苏联的主导阵营只接收被直接点名的目标国，不整块拖拽
                ROOT = { leave_faction = yes }
                PAU = { faction_leader = { add_to_faction = ROOT } }
            }
            else = {
                ROOT = {
                    every_faction_member = {
                        display_individual_scopes = yes
                        leave_faction = yes
                        PAU = {
                            faction_leader = {
                                #PREV 是 PAU，PREV.PREV 是当前阵营成员
                                add_to_faction = PREV.PREV
                            }
                        }
                    }
                }
            }
        }
        else = {
            PAU = { faction_leader = { add_to_faction = ROOT } }
        }
    }
    ROOT = { clr_country_flag = PAU_allow_major_faction_pull }
}
