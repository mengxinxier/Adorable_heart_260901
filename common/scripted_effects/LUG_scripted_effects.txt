	if = {
		limit = {
			check_variable = {
				LUG_pub_slot_A = 0
			}
		}
		set_variable = {
			LUG_pub_slot_A = LUG_pub_temp
		}
	}
}

LUG_Subaru_mental_state_check_effect = {
	hidden_effect = {
		if = {
			limit = {
				AND = {
                    check_variable = { var = Subaru_mental_state value = 80 compare = greater_than_or_equals }
                    check_variable = { var = Subaru_mental_state value = 101 compare = less_than }
                }
			}
			remove_ideas = LUG_Subaru_idea_3
            remove_ideas = LUG_Subaru_idea_2
            remove_ideas = LUG_Subaru_idea_1
            add_ideas = LUG_Subaru_idea
		}
		if = {
			limit = {
                AND = {
                    check_variable = { var = Subaru_mental_state value = 50 compare = greater_than_or_equals }
                    check_variable = { var = Subaru_mental_state value = 80 compare = less_than }
                }
			}
			remove_ideas = LUG_Subaru_idea_3
            remove_ideas = LUG_Subaru_idea_2
            remove_ideas = LUG_Subaru_idea
            add_ideas = LUG_Subaru_idea_1
		}
		else_if = {
			limit = {
				AND = {
                    check_variable = { var = Subaru_mental_state value = 20 compare = greater_than_or_equals }
                    check_variable = { var = Subaru_mental_state value = 50 compare = less_than }
                }
			}
			remove_ideas = LUG_Subaru_idea_3
            remove_ideas = LUG_Subaru_idea
            remove_ideas = LUG_Subaru_idea_1
            add_ideas = LUG_Subaru_idea_2
		}
		else_if = {
			limit = {
				AND = {
                    check_variable = { var = Subaru_mental_state value = 0 compare = greater_than_or_equals }
                    check_variable = { var = Subaru_mental_state value = 20 compare = less_than }
                }
			}
			remove_ideas = LUG_Subaru_idea
            remove_ideas = LUG_Subaru_idea_2
            remove_ideas = LUG_Subaru_idea_1
            add_ideas = LUG_Subaru_idea_3
		}
	}
}
