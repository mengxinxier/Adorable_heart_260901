GRE_enable_new_tarot = {
	set_variable = { GRE_enable_tarot_id = temp_enable_tarot_id }
	add_to_variable = { GRE_enable_tarot_id = 15 }
	for_each_loop = {
		array = GRE_Iuno_Tarot_pie_chart_array
		index = i
		value = v
		break = break
		
		if = {
			limit = {
				check_variable = { GRE_Iuno_Tarot_pie_chart_array^i = GRE_enable_tarot_id }
			}
			subtract_from_variable = { GRE_Iuno_Tarot_pie_chart_array^i = 15 }
		}
	}
}
