##检测重复
check_repetitive_compatible_culture = {
	clear_array = compatible_culture_array_temp
	for_each_loop = {
		array = compatible_culture_array
		value = current_culture
		index = current_index
		# 检查当前元素是否不在临时数组中
		if = {
			limit = {
				NOT = { is_in_array = { array = compatible_culture_array_temp value = current_culture } }
			}
			# 不在则添加到临时数组
			add_to_array = { compatible_culture_array_temp = current_culture }
		}
	}
	clear_array = compatible_culture_array
	for_each_loop = {
		array = compatible_culture_array_temp
		value = current_culture
		index = current_index
		add_to_array = { compatible_culture_array = current_culture }
	}
	clear_array = compatible_culture_array_temp
}
