			}
			set_global_flag = JAP_Fading_Sun
			limit = {
				NOT = { has_global_flag = JAP_zhongrizhanzhengshibai }
			}
			set_global_flag = JAP_zhongrizhanzhengshibai
			JAP = {
				add_timed_idea = { idea = PRC_Sunset days = 365 }
				#清除了残余战争借口
				every_enemy_country = {
					JAP = {
						remove_wargoal = {
							type = all
							target = PREV
						}
					}
				}
				#附庸国独立（中国及朝鲜）
				if = {
					limit = { exists = yes }
					every_country = {
						limit = {
							OR = {
								is_guaranteed_by = PREV
								is_subject_of = PREV
								has_non_aggression_pact_with = PREV
								has_military_access_to = PREV
								gives_military_access_to = PREV
							}
							or = {
								is_chinese_country = yes
								tag = KOR
							}
						}
						diplomatic_relation = {
							country = JAP
							relation = military_access
							active = no
						}
						JAP = {
							if = {
								limit = { has_subject = PREV }
								set_autonomy = {
									target = PREV
									autonomy_state = autonomy_free
								}
							}
							diplomatic_relation = {
								country = PREV
								relation = guarantee
								active = no
							}
							diplomatic_relation = {
								country = PREV
								relation = non_aggression_pact
								active = no
							}
							diplomatic_relation = {
								country = PREV
								relation = military_access
								active = no
							}
						}
					}
				}
			}
			#清除本土外的核心与宣称（国区）
			every_state = {
				limit = { has_state_flag = is_whole_chinese_state_area }
				remove_core_of = JAP
				remove_claim_by = JAP

China_tribute_system_effect = {
	###  统计经济阵营成员数量
	clear_array = global.China_tribute_system_member  ##成员所在的数组
	clear_array = global.CTS_Imperial_Province_member
	clear_array = global.CTS_Imperial_Vassal_member
	clear_array = global.CTS_Imperial_Tributary_member
	set_variable = { global.num_of_China_tribute_system = 0 }  ##成员数量变量
	every_country = {
		limit = { check_variable = { THIS.economy_faction_var = 7 } }
		add_to_array = { global.China_tribute_system_member = THIS }
		add_to_variable = { global.num_of_China_tribute_system = 1 }
		if = {
			limit = { has_autonomy_state = China_Provinces }
			add_to_array = { global.CTS_Imperial_Province_member = THIS }
		}
		else_if = {
			limit = { has_autonomy_state = China_Domain }
			add_to_array = { global.CTS_Imperial_Province_member = THIS }
		}
		else_if = {
			limit = { has_autonomy_state = China_Vassal }
			add_to_array = { global.CTS_Imperial_Vassal_member = THIS }
		}
		else_if = {
			limit = {
				is_subject = no
				not = { tag = CHI }
			}
			add_to_array = { global.CTS_Imperial_Tributary_member = THIS }
		}
	}
	###  计算动态阵营buff（如果有）
	China_tribute_system_leader_modifier_effect = yes
	###  刷新经济阵营GDP
	compute_global_gdp = yes
}
