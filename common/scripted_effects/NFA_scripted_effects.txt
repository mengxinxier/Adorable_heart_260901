###  美洲经济同盟计算  ###
American_Economic_Union_effect = {
	clear_array = global.American_Economic_Union_member
	set_variable = { global.num_of_American_Economic_Union = 0 }
	every_country = {
		limit = { check_variable = { THIS.economy_faction_var = 9 } }
		add_to_array = { global.American_Economic_Union_member = THIS }
		add_to_variable = { global.num_of_American_Economic_Union = 1 }
	}
	###  刷新经济阵营GDP
	compute_global_gdp = yes
}
