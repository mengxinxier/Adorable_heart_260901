# State scope; PREV must be the integrating country (e.g. ROOT in annexation events).
# 北美命运事件整合保底：同语种50%顺从度造核，否则80%顺从度造核。
NA_enable_state_integration_by_language = {
	if = {
		limit = {
			NOT = { is_core_of = PREV }
		}
		if = {
			limit = {
				NA_is_same_language_state_as_PREV = yes
			}
			enable_state_integration_50_compliance = yes
		}
		else = {
			enable_state_integration_80_compliance = yes
		}
	}
}
