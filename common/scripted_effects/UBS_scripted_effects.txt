###  刷新收入-工业及消费税
AHES_Refresh_income_indus = {
    ##工业税
    set_variable = { AHES_Industrial_tax_income = num_of_civilian_factories_available_for_projects }  ##工业税
    multiply_variable = { AHES_Industrial_tax_income = 1.7 }
    set_variable = { temp_AHES_Industrial_tax_income = 1 }
    add_to_variable = { temp_AHES_Industrial_tax_income = modifier@hcp_Industrial_tax_factor_monthly }
    multiply_variable = { AHES_Industrial_tax_income = temp_AHES_Industrial_tax_income }
    round_variable = AHES_Industrial_tax_income
    ##消费税
    set_variable = { AHES_num_of_CGCF = num_of_civilian_factories }  ##消费品民工
    subtract_from_variable = { AHES_num_of_CGCF = num_of_civilian_factories_available_for_projects }
    set_variable = { AHES_Consumption_tax_income = AHES_num_of_CGCF }  ##消费税收入
    multiply_variable = { AHES_Consumption_tax_income = 2.2 }
    set_variable = { temp_AHES_Consumption_tax_income = 1 }
    add_to_variable = { temp_AHES_Consumption_tax_income = modifier@hcp_Consumption_tax_factor_monthly }
    multiply_variable = { AHES_Consumption_tax_income = temp_AHES_Consumption_tax_income }
    round_variable = AHES_Consumption_tax_income
    ##添加到总收入中
    add_to_variable = { AHES_Total_income = AHES_Industrial_tax_income }
    add_to_variable = { AHES_Total_income = AHES_Consumption_tax_income }
}

###  刷新收入-外交收入修正
AHES_Refresh_income_dipol_factor = {
    set_variable = { AHES_diplo_income_correction_coefficient = 0 }
    set_variable = { temp_income_multiply_factor = 0 }  ##临时乘数
    set_variable = { temp_outcome_for_overlord = 0 }  ##临时支出-贡献给宗主国
    set_variable = { temp_outcome_for_CHI = 0 }  ##临时支出-贡献给CHI
    set_variable = { AHES_diplo_outcome = 0 }  ##支出-贡献给宗主国
    if = {
        limit = { is_subject_of = CHI }
        add_to_variable = { temp_income_multiply_factor = modifier@hcp_income_for_overlord_factor_monthly }
        add_to_variable = { temp_income_multiply_factor = modifier@hcp_income_for_CHI_factor_monthly }
        subtract_from_variable = { AHES_diplo_income_correction_coefficient = modifier@hcp_income_for_overlord_factor_monthly }
        subtract_from_variable = { AHES_diplo_income_correction_coefficient = modifier@hcp_income_for_CHI_factor_monthly }
        set_variable = { temp_outcome_for_overlord = AHES_Total_income }
        multiply_variable = { temp_outcome_for_overlord = temp_income_multiply_factor }
        round_variable = temp_outcome_for_overlord
        round_variable = temp_outcome_for_CHI
        CHI = { add_to_variable = { AHES_diplo_exincome = PREV.temp_outcome_for_overlord } }
    }
    else_if = {
        limit = { is_subject = yes }
        add_to_variable = { temp_income_multiply_factor = modifier@hcp_income_for_overlord_factor_monthly }
        subtract_from_variable = { AHES_diplo_income_correction_coefficient = modifier@hcp_income_for_overlord_factor_monthly }
        set_variable = { temp_outcome_for_overlord = AHES_Total_income }
        multiply_variable = { temp_outcome_for_overlord = temp_income_multiply_factor }
        round_variable = temp_outcome_for_overlord
        round_variable = temp_outcome_for_CHI
        overlord = { add_to_variable = { AHES_diplo_exincome = PREV.temp_outcome_for_overlord } }
    }
    else_if = {
        limit = { CHI = { is_subject = no } }
        add_to_variable = { temp_income_multiply_factor = modifier@hcp_income_for_CHI_factor_monthly }
        subtract_from_variable = { AHES_diplo_income_correction_coefficient = modifier@hcp_income_for_CHI_factor_monthly }
        set_variable = { temp_outcome_for_CHI = AHES_Total_income }
        multiply_variable = { temp_outcome_for_CHI = temp_income_multiply_factor }
        round_variable = temp_outcome_for_overlord
        round_variable = temp_outcome_for_CHI
        CHI = { add_to_variable = { AHES_diplo_exincome = PREV.temp_outcome_for_CHI } }
    }
    add_to_variable = { AHES_diplo_outcome = temp_outcome_for_overlord }
    add_to_variable = { AHES_diplo_outcome = temp_outcome_for_CHI }
    subtract_from_variable = { AHES_Total_income = temp_outcome_for_overlord }
    subtract_from_variable = { AHES_Total_income = temp_outcome_for_CHI }
}

###  刷新收入-额外收入

    set_variable = { AHES_other_exincome = exincome }

compute_global_gdp = {
    set_variable = { global.global_gdp_var = 0 } #所有国家gdp
    set_variable = { global.temp_economy_faction_index = 1 } #临时变量
    clear_array = global.economy_faction_GDP  ##各个经济阵营GDP
    clear_array = global.economy_faction_GDP_ratio  ##各个经济阵营GDP占比
    clear_array = global.economy_faction_GDP_temp  ##各个经济阵营GDP副本(仅用于排序)
    clear_array = global.economy_faction_GDP_rank  ##各个经济阵营GDP排行
    clear_array = global.economy_faction_GDP_rank_num  ##各个经济阵营GDP排行数字
    for_each_scope_loop = { #GDP归零
        array = global.econ_countries
        if = {
            limit = { check_variable = { THIS.economy_faction_var > 0 } }
            set_variable = { global.economy_faction_GDP^economy_faction_var = 0 }
        }
    }
    for_each_scope_loop = { #计算所有国家gdp及阵营GDP
        array = global.econ_countries
        add_to_variable = { global.global_gdp_var = AHES_Total_income }
        if = {
            limit = { check_variable = { THIS.economy_faction_var > 0 } }
            add_to_variable = { global.economy_faction_GDP^economy_faction_var = AHES_Total_income }
            add_to_variable = { global.economy_faction_GDP_temp^economy_faction_var = AHES_Total_income }
        }
    }
    for_each_scope_loop = { #计算阵营GDP占比与排名
        array = global.econ_countries
        set_variable = { global.economy_faction_GDP_ratio^global.temp_economy_faction_index = global.economy_faction_GDP^global.temp_economy_faction_index }
        divide_variable = { global.economy_faction_GDP_ratio^global.temp_economy_faction_index = global.global_gdp_var }
        add_to_variable = { global.temp_economy_faction_index = 1 }
    }
    clear_array = global.economy_faction
    set_variable = { global.economy_faction_index = 0 }
    for_each_scope_loop = {
        array = global.econ_countries
        if = {
            limit = {
                check_variable = { THIS.economy_faction_var > 0 }
                not = { is_in_array = { global.economy_faction_GDP_rank = THIS.economy_faction_var } }
            }
            add_to_array = { global.economy_faction_GDP_rank = THIS.economy_faction_var }
            add_to_array = { global.economy_faction = THIS }
            set_variable = { economy_faction_temp_GDP = global.economy_faction_GDP^economy_faction_var }
        }
    }
    get_sorted_scored_countries = {
        scorer = economy_faction_rank_scorer
        array = global.economy_faction
    }
    for_each_scope_loop = {
        array = global.economy_faction
        add_to_variable = { global.economy_faction_index = 1 }
        set_variable = { global.economy_faction_GDP_rank_num^economy_faction_var = global.economy_faction_index }
    }
    clear_variable = global.economy_faction_index
    Wto_leader_modifier_effect = yes
}

econmic_debuff_calculation = {
    if = {
        limit = { check_variable = { AHES_Total_income > 1 } }
        set_variable = { this_country_GDP = AHES_Total_income }
    }
    else = {
        set_variable = { this_country_GDP = 1 }
    }
    multiply_variable = {
        var = this_country_GDP
        value = 12
    }
    set_variable = {
        Debt_ratio = ubs_load
    }
    divide_variable = {
        var = Debt_ratio
        value = this_country_GDP
    }
    if = {
        limit = { check_variable = { Debt_ratio > 1 } }
        set_variable = {
            Debt_ratio_2 = Debt_ratio
        }
        add_to_variable = {
            Debt_ratio_2 = -1
        }
        set_variable = {
            UBS_debuff_stability_factor = Debt_ratio_2
        }
        divide_variable = {
            var = UBS_debuff_stability_factor
            value = -4
        }
        set_variable = {
            UBS_debuff_production_speed_buildings_factor = UBS_debuff_stability_factor
        }
        set_variable = {
            UBS_debuff_industrial_capacity_factory = Debt_ratio_2
        }
        divide_variable = {
            var = UBS_debuff_industrial_capacity_factory
            value = -10
        }
        clamp_variable = {
            var = UBS_debuff_stability_factor
            min = -0.8
            max = 0
        }
        clamp_variable = {
            var = UBS_debuff_production_speed_buildings_factor
            min = -0.8
            max = 0
        }
        clamp_variable = {
            var = UBS_debuff_industrial_capacity_factory
            min = -0.8
            max = 0
        }
    }
    else = {
        set_variable = { UBS_debuff_production_speed_buildings_factor = 0 }
        set_variable = { UBS_debuff_industrial_capacity_factory = 0 }
        set_variable = { UBS_debuff_stability_factor = 0 }
    }
    if = {
        limit = {
            check_variable = { Debt_ratio > 1 }
            not = { has_dynamic_modifier = { modifier = econmic_debuff_modifer } }
        }
        if = {
            limit = { is_ai = yes }
            add_ideas = AI_Arms_imitation
        }
        hidden_effect = { country_event = generic.2026 }
        add_dynamic_modifier = { modifier = econmic_debuff_modifer }
    }
    if = {
        limit = {
            check_variable = {
                Debt_ratio = 1
            }
            has_dynamic_modifier = {
                modifier = econmic_debuff_modifer
            }
        }
        if = {
            limit = {
                has_idea = AI_Arms_imitation
            }
            remove_ideas = AI_Arms_imitation
        }
        remove_dynamic_modifier = {
            modifier = econmic_debuff_modifer
        }
    }
    else_if = {
        limit = {
            check_variable = {
                Debt_ratio < 1
            }
            has_dynamic_modifier = {
                modifier = econmic_debuff_modifer
            }
        }
        if = {
            limit = {
                has_idea = AI_Arms_imitation
            }
            remove_ideas = AI_Arms_imitation
        }
        remove_dynamic_modifier = {
            modifier = econmic_debuff_modifer
        }
    }
}

    AHES_Refresh_income_and_outcome = yes
    if = {
        limit = { check_variable = { ubs_save < 0 } }
        subtract_from_variable = {
            var = ubs_load
            value = ubs_save
        }
        clamp_variable = {
            var = ubs_save
            min = 0
        }
    }
    if = {
        limit = { check_variable = { ubs_load < 0 } }
        subtract_from_variable = {
            var = ubs_save
            value = ubs_load
        }
        clamp_variable = {
            var = ubs_load
            min = 0
        }
    }
    clamp_variable = {
        var = THIS.ubs_load
        min = 0
    }
    clamp_variable = {
        var = ubs_save
        min = 0
    }
    round_variable = THIS.ubs_load
    round_variable = ubs_save
    every_country = {
        limit = { NOT = { has_country_flag = shouru_zhichu_CD_30 } }
        ##结算存款与债务
        if = {
            limit = {
                #收入为正
                check_variable = { AHES_Net_income > 0 }
            }
            add_to_variable = {
                var = ubs_save
                value = AHES_Net_income
            }
        }
        else = {
            #收入为0或负
            subtract_from_variable = {
                var = ubs_load
                value = AHES_Net_income
            }
        }
        ###自动还款###
        if = {
            limit = {
                check_variable = { ubs_save > 0 }
                check_variable = { ubs_load > 0 }
                check_variable = { adorable_heart_automatic_repayment_var = 2 }
            }
            if = {
                limit = {
                    not = { check_variable = { ubs_load > ubs_save } }
                }
                subtract_from_variable = { ubs_save = ubs_load }
                set_variable = { ubs_load = 0 }
            }
            else = {
                subtract_from_variable = { ubs_load = ubs_save }
                set_variable = { ubs_save = 0 }
            }
        }
        #####利息结算####
        multiply_variable = {
            var = ubs_load

                not = { check_variable = { AHES_Net_income < 0 } }

                not = { check_variable = { current_jinglirun_icon_frame = 1 } }
            }
            set_variable = { current_jinglirun_icon_frame = 1 }

        set_country_flag = {
            flag = shouru_zhichu_CD_30

        }
