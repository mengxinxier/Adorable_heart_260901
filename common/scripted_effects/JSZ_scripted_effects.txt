JSZ_national_update = {
		clamp_variable = { var = JSZ_national_influence_var min = 0 max = 100 }
		hidden_effect = {
			JSZ_remove_national_idea = yes
			if = {
				limit = { not = { check_variable = { JSZ_national_influence_var > 20 } } }
				add_ideas = JSZ_national_idea_1
			}
			else_if = {
				limit = {
					check_variable = { JSZ_national_influence_var > 20 }
					not = { check_variable = { JSZ_national_influence_var > 40 } }
				}
				add_ideas = JSZ_national_idea_2
			}
			else_if = {
				limit = { 
					check_variable = { JSZ_national_influence_var > 40 }
					not = { check_variable = { JSZ_national_influence_var > 60 } }
				}
				add_ideas = JSZ_national_idea_3
			}
			else_if = {
				limit = { 
					check_variable = { JSZ_national_influence_var > 60 }
					not = { check_variable = { JSZ_national_influence_var > 80 } }
				}
				add_ideas = JSZ_national_idea_4
			}			
			else_if = {
				limit = { 
					check_variable = { JSZ_national_influence_var > 80 }
					not = { check_variable = { JSZ_national_influence_var > 100 } }
				}
				add_ideas = JSZ_national_idea_5
			}
		}
	}
JSZ_remove_national_idea = {
			remove_ideas = JSZ_national_idea_1
			remove_ideas = JSZ_national_idea_2
			remove_ideas = JSZ_national_idea_3
			remove_ideas = JSZ_national_idea_4
			remove_ideas = JSZ_national_idea_5
	}
