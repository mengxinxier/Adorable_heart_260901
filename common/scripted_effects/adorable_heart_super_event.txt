#ordinary
adorable_heart_fire_super_event = {
    if = {
        limit = {
            not = { check_variable = { ROOT.adorable_heart_super_event_var = 0 } }
            not = { is_in_array = { global.superevents_note_array = ROOT.adorable_heart_super_event_var } }
        }
        add_to_array = { global.superevents_note_array = ROOT.adorable_heart_super_event_var }
        add_to_array = { global.superevents_note_date_array = global.date }
    }
    hidden_effect = {
        if = {
            limit = {
                check_variable = {
                    adorable_heart_super_event_music_button_var = 2
                }
            }
            adorable_heart_fire_super_event_music = yes
        }
    }
}

SOV_DVL_WAR_super_event = {
    set_variable = { adorable_heart_super_event_var = 5 }
    adorable_heart_fire_super_event = yes
}

#changable
adorable_heart_clr_super_event_flag = {
    clear_variable = adorable_heart_super_event_var
}

##USE SUPER_EVENT_MAKER_GUI IN USEFUL PYTHON PROGRAMS TO MAKE NEW SUPER EVENTS##
adorable_heart_fire_super_event_music = {
    hidden_effect = {
		meta_effect = {
			text = {
                scoped_play_song = "super_event_[super_event_var]"
			}
			super_event_var = "[?adorable_heart_super_event_var]"
		}
    }
}
