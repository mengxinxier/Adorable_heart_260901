TAM_bop_mid = {
	if = {
		limit = {
			has_power_balance = {
				id = TAM_communism_bop
			}
		}
		if = {
			limit = { 
				power_balance_value = {
					id = TAM_communism_bop
					value > 0
				}
			}
			add_power_balance_value = {
				id = TAM_communism_bop
				value = -0.02
			}
		}
		else = {
			add_power_balance_value = {
				id = TAM_communism_bop
				value = 0.02
			}
		}
	}
}
