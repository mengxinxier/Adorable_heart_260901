CAC_SOV_intervention_change_value = {
	
	set_variable = { CAC_SOV_intervention_temp_sum_value = CAC_SOV_intervention_icon_value } 
	add_to_variable = { 
		CAC_SOV_intervention_temp_sum_value = CAC_SOV_intervention_value_to_add
	}


	if = { ## If the new value reaches 100
		limit = {
			check_variable = {
				var = CAC_SOV_intervention_temp_sum_value
				value = 100
				compare = greater_than_or_equals
			}
		}
		CAC_SOV_intervention_max_value_reached = yes
	}
	else_if = {
		limit = {
			check_variable = {
				var = CAC_SOV_intervention_temp_sum_value
				value = 0
				compare = less_than_or_equals
			}
		}
		### Is is less than 0
		CAC_SOV_intervention_min_value_reached = yes
	}
	add_to_variable = {
		CAC_SOV_intervention_icon_value = CAC_SOV_intervention_value_to_add
	}
	clamp_variable = {
		var = CAC_SOV_intervention_icon_value
		min = 0
		max = 100
	}
}

CAC_SOV_intervention_max_value_reached = {
	hidden_effect = {
		every_country = {
			limit = {
				exists = yes
				has_cosmetic_tag = Tresen_Federation
			}
			set_country_flag = CAC_SOV_intervention_is_over ## Removes decision category
		}
		SOV = {
			country_event = {
				id = TAM.13
			}
			set_country_flag = CAC_SOV_wants_more
		}
	}

}

CAC_SOV_intervention_min_value_reached = {
	#custom_effect_tooltip = SOV_not_wants_more
	hidden_effect = {
		every_country = {
			limit = {
				exists = yes
				has_cosmetic_tag = Tresen_Federation
			}
			set_country_flag = CAC_SOV_intervention_is_over ## Removes decision category

			}
		}

		}

	}
	
	
}
