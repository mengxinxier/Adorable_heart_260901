TAM_For_the_Mortal_ex = {
	if = {
		limit = {
			tag = TAM
			is_ai = no
			has_war_with_major = yes
			surrender_progress > 0.80
			date > 1941.7.1
			NOT = {
				has_country_flag = TAM_For_the_Mortal
			}
		}
		set_country_flag = TAM_For_the_Mortal
		mark_focus_tree_layout_dirty = yes
	}
}

TAM_champion_slot_set = {
	if = {
		limit = { 
			check_variable = {
				TAM_champion_slot_A = 0
			}
		}
		set_variable = {
			TAM_champion_slot_A = TAM_champion_temp
		}
	}
	else_if = {
		limit = { 
			check_variable = {
				TAM_champion_slot_B = 0
			}
		}
		set_variable = {
			TAM_champion_slot_B = TAM_champion_temp
		}
	}
	else_if = {
		limit = { 
			check_variable = {
				TAM_champion_slot_C = 0
			}
		}
		set_variable = {
			TAM_champion_slot_C = TAM_champion_temp
		}
	}
}
TAM_champion_slot_clr = {
	if = {
		limit = { 
			check_variable = {
				TAM_champion_slot_A = TAM_champion_temp
			}
		}
		set_variable = {
			TAM_champion_slot_A = 0
		}
	}
	else_if = {
		limit = { 
			check_variable = {
				TAM_champion_slot_B = TAM_champion_temp
			}
		}
		set_variable = {
			TAM_champion_slot_B = 0
		}
	}
	else_if = {
		limit = { 
			check_variable = {
				TAM_champion_slot_C = TAM_champion_temp
			}
		}
		set_variable = {
			TAM_champion_slot_C = 0
		}
	}
}
TAM_champion_slot_unlock = {
	custom_effect_tooltip = TAM_tt_champion_slot_unlock
	if = {
		limit = { 
			check_variable = {
				TAM_champion_slot_B = -1
			}
		}
		set_variable = {
			TAM_champion_slot_B = 0
		}
	}
	else = {
		set_variable = {
			TAM_champion_slot_C = 0
		}
	}
}
