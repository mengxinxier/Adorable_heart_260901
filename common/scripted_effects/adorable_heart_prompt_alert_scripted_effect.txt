adorable_heart_prompt_alert_operation = {
	if = {
		limit = {
            root = { is_ai = no }
        }
		hidden_effect = {
			scoped_sound_effect = "diplomatic_notification"
			add_to_array = { adorable_heart_prompt_alerts = adorable_heart_prompt_alert_operations^0 }
			add_to_array = { adorable_heart_prompt_alert_other_var1 = adorable_heart_prompt_alert_operations^1 }
			add_to_array = { adorable_heart_prompt_alert_other_var2 = adorable_heart_prompt_alert_operations^2 }
			add_to_array = { adorable_heart_prompt_alert_other_var3 = adorable_heart_prompt_alert_operations^3 }
			add_to_array = { adorable_heart_prompt_alert_other_var4 = adorable_heart_prompt_alert_operations^4 }
			add_to_array = { adorable_heart_prompt_alert_other_var5 = adorable_heart_prompt_alert_operations^5 }
	
			if = {
				limit = {
					has_variable = temp_var:temp_tag
				}
				add_to_array = { adorable_heart_prompt_alert_flags = temp_tag }
			}
			else = {
				add_to_array = { adorable_heart_prompt_alert_flags = ROOT }
			}
			if = {
				limit = {
					has_variable = temp_var:temp_tag2
				}
				add_to_array = { adorable_heart_prompt_alert_flags2 = temp_tag2 }
			}
			else = {
				add_to_array = { adorable_heart_prompt_alert_flags2 = ROOT }
			}
	
			add_to_array = { adorable_heart_prompt_alerts_progressbars = -317 }
			set_variable_to_random = global.adorable_heart_alerts_dirty
	
			clear_array = adorable_heart_prompt_alert_operations
	
			country_event = { id = alerts_mechanism.1 }
		}
	}
}
