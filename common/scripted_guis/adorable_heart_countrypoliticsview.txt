scripted_gui = {
	another_country_leader_view = {
		context_type = player_context
		window_name = "another_country_leader_view"
		parent_window_token = politics_tab
		properties = {
			country_leader_picture = {
				image = "[Country_leader_picture]"
			}
			pol_view_bg = {
				image = "[pol_view_bg_picture]"
			}
		}
		triggers = {
			AH_debug_buff_rank_visible = { is_debug = yes }
			AH_debug_se_up_visible = { is_debug = yes tag = FOT }
			AH_debug_se_down_visible = { is_debug = yes tag = FOT }
			AH_debug_se_music_visible = { is_debug = yes tag = FOT }
			AH_debug_se_text_visible = { is_debug = yes tag = FOT }
		}
		visible = {
			always = yes
		}
		effects = {
			AH_debug_buff_rank_click = {
				if = {
					limit = {
						not = { has_country_flag = Adorable_Heart_buff_rank_window_visible }
					}
					set_country_flag = Adorable_Heart_buff_rank_window_visible
				}
				clear_array = global.countries_buff
				set_variable = { global.countries_buff_index = 0 }
				every_country = {
					set_variable = { temp_rank_buff_var = num_of_factories }
					add_to_array = {
						global.countries_buff = THIS
					}
				}
				get_sorted_scored_countries = {
					scorer = Buff_rank_scorer
					array = global.countries_buff
				}
				# Set up ranking positions
				for_each_scope_loop = {
					array = global.countries_buff
					add_to_variable = {
						global.countries_buff_index = 1
					}
					set_variable = {
						econ_ranking = global.countries_buff_index
					}
				}
				clear_variable = global.countries_buff_index
			}
			AH_debug_se_up_click = {
				add_to_variable = { adorable_heart_super_event_var = 1 }
			}
			AH_debug_se_down_click = {
				add_to_variable = { adorable_heart_super_event_var = -1 }
			}
			AH_debug_se_music_click = {
				adorable_heart_fire_super_event = yes
			}
			AH_debug_se_up_control_click = {
				add_to_variable = { adorable_heart_super_event_var = 10 }
			}
			AH_debug_se_down_control_click = {
				add_to_variable = { adorable_heart_super_event_var = -10 }
			}
		}
	}
}
