			GRE_TDT_category_type = {
				frame = GRE_TDT_category_type_frame
			}
		}
		effects = {
			GRE_TDT_map_state_click = {
				ROOT = {
					add_political_power = 2
					for_each_loop = {
						array = GRE_TDT_map_state_array
						index = i
						value = v
						break = b
						
						set_variable = { GRE_TDT_map_state_array^i = 1 }
						if = {
							limit = {
								check_variable = { i = si }
							}
							set_variable = { GRE_TDT_map_state_array^i = 2 }
						}
					}
					GRE_TDT_state_id_from_map_to_decision = yes
					GRE_TDT_refresh_map = yes
				}
			}
			GRE_TDT_category_map_click = {
				ROOT = {
					set_variable = { GRE_TDT_category = 1 }
					set_variable = { GRE_TDT_category_map_frame = 2 }
					set_variable = { GRE_TDT_category_type_frame = 1 }
					GRE_TDT_refresh_map = yes
				}
			}
			GRE_TDT_category_type_click = {
				ROOT = {
					set_variable = { GRE_TDT_category = 2 }
					set_variable = { GRE_TDT_category_map_frame = 1 }
					set_variable = { GRE_TDT_category_type_frame = 2 }
					GRE_TDT_refresh_map = yes
				}
			}
		}
